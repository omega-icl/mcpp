538template <class T> inline T
539Specbnd<T>::_LambdaS
540( const fadbad::F<T> &a, unsigned int n )
541{
542 if( !n ) return 0.;
543 if( n == 1 ) return a.size()? Op<T>::sqr( a[0] ): 0.;
544 double upbnd=0.;
545 for (unsigned int i=0; i<n; i++)
546 if( a.size() ) upbnd += Op<T>::u( Op<T>::sqr( a[i] ) );
547 return Op<T>::zeroone() * upbnd;
548}
549
550template <class T> inline T
551Specbnd<T>::_LambdaT
552( const fadbad::F<T> &a, const fadbad::F<T> &b, unsigned int n )
553{
554 if( !n ) return 0.;
555 if( n == 1 ) return (a.size() && b.size())? 2.*a[0]*b[0]: 0.;
556 double upbnda=0., upbndb=0.;
557 for (unsigned int i=0; i<n; i++){
558 if( a.size() ) upbnda += Op<T>::u( Op<T>::sqr( a[i] ) );
559 if( b.size() ) upbndb += Op<T>::u( Op<T>::sqr( b[i] ) );
560 }
561 T lamb = 2.*(Op<T>::zeroone()-0.5) * std::sqrt(upbnda*upbndb);
562 for (unsigned int i=0; i<n; i++)
563 if( a.size() && b.size() ) lamb += a[i]*b[i];
564 return lamb;
565}

567template <typename T> inline std::pair<double,double>
568Specbnd<T>::spectrum
569( const fadbad::B< fadbad::F< double > >* D2X )
570{
571 if( !D2X ) throw typename Specbnd<T>::Exceptions( Specbnd<T>::Exceptions::SPECTR );
572 const unsigned int N = D2X->val().size();
573 if( !N ) throw typename Specbnd<T>::Exceptions( Specbnd<T>::Exceptions::SPECTR );
574 double*H = new double[N*N];
575
576 for( unsigned int j=0; j<N; j++ )
577 for( unsigned int i=j; i<N; i++ )
578 H[j*N+i] = H[i*N+j] = ( i==j? D2X[i].deriv(0).deriv(i):
579 0.5*(D2X[i].deriv(0).deriv(j)+D2X[j].deriv(0).deriv(i)) );
580#ifdef MC__SPECBND_DEBUG_SPECTRUM
581 mc::display( N, N, H, N, "\nMatrix H", std::cout );
582#endif
583
584 double*D = mc::dsyev_wrapper( N, H );
585 delete[] H;
586 if( !D ) throw typename Specbnd<T>::Exceptions( Specbnd<T>::Exceptions::SPECTR );
587 std::pair<double,double> spbnd = std::make_pair( min(N,D), max(N,D) );
588 delete[] D;
589 return spbnd;
590}

638template <typename T> inline std::pair<double,double>
639Specbnd<T>::spectral_bound
640( const fadbad::B< fadbad::F< T > >* D2F )
641{
642 if( !D2F ) throw typename Specbnd<T>::Exceptions( Specbnd<T>::Exceptions::HESSBND );
643 const unsigned int N = D2F->val().size();
644 if( !N ) throw typename Specbnd<T>::Exceptions( Specbnd<T>::Exceptions::HESSBND );
645 T pD2F[N*N];
646 for( unsigned int i=0, k=0; i<N; i++ )
647 for( unsigned int j=0; j<N; j++, k++ )
648 pD2F[k] = D2F[i].deriv(0).deriv(j);
649
650 switch( options.HESSBND ){
651 case Options::GERSHGORIN:
652 return _gershgorin_bound( N, pD2F );
653 case Options::HERTZROHN: default:
654 return _hertzrohn_bound( N, pD2F );
655 }
656}

678Specbnd<T>::spectral_bound_re
679( const unsigned N, const T*A )
680{

682#ifdef MC__SPECBND_DEBUG_HESSBND

688 ARe[i+N*j] = ( A[i+N*j] + A[i*N+j] ) / 2.;
689#ifdef MC__SPECBND_DEBUG_HESSBND
690 mc::display( N, N, ARe, N, "\nMatrix ARe", std::cout );
691#endif
692 return spectral_bound( N, ARe );

695template <typename T> inline std::pair<double,double>
696Specbnd<T>::spectral_bound_im
697( const unsigned N, const T*A )
698{
699 if( !N || !A ) throw typename Specbnd<T>::Exceptions( Specbnd<T>::Exceptions::HESSBND );
700 const unsigned N2 = 2*N;
701#ifdef MC__SPECBND_DEBUG_HESSBND
702 mc::display( N, N, A, N, "\nMatrix A", std::cout );
703#endif
704 T AIm[N2*N2];
705 for( unsigned int i=0; i<N; i++ )
706 for( unsigned int j=0; j<N; j++ ){
707 AIm[i+N2*j] = AIm[N2*N+N+i+N2*j] = 0.;
708 AIm[N+i+N2*j] = ( A[i+N*j] - A[i*N+j] ) / 2.;
709 AIm[N2*N+i+N2*j] = - AIm[N+i+N2*j];
710 }
711#ifdef MC__SPECBND_DEBUG_HESSBND
712 mc::display( N2, N2, AIm, N2, "\nMatrix AIm", std::cout );
713#endif
714 return spectral_bound( N2, AIm );

730template <typename T> inline std::pair<double,double>
731Specbnd<T>::_gershgorin_bound
732( const unsigned N, const T*D2F )
733{
734 struct ndx{
735 static unsigned cw( const unsigned i, const unsigned j, const unsigned n ) { return i+j*n; }
736 static unsigned rw( const unsigned i, const unsigned j, const unsigned n ) { return i*n+j; }
737 };
738 std::pair<double,double> spbnd;
739
740 for( unsigned int i=0; i<N; i++ ){
741 double ri = 0.;
742 for( unsigned int j=0; j<N; j++ ){
743 if( j == i ) continue;
744 T D2Fij;
745 if( !Op<T>::inter( D2Fij, D2F[ndx::rw(i,j,N)], D2F[ndx::rw(j,i,N)] ) )
746 D2Fij = D2F[ndx::rw(i,j,N)];
747 ri += Op<T>::abs( D2Fij );
748 }
749
750 if( !i ){
751 spbnd.first = Op<T>::l(D2F[ndx::rw(i,i,N)]) - ri;
752 spbnd.second = Op<T>::u(D2F[ndx::rw(i,i,N)]) + ri;
753 }
754 else{
755 spbnd.first = std::min( spbnd.first, Op<T>::l(D2F[ndx::rw(i,i,N)]) - ri );
756 spbnd.second = std::max( spbnd.second, Op<T>::u(D2F[ndx::rw(i,i,N)]) + ri );
757 }
758 }
759
760 return spbnd;
761}

763template <typename T> inline std::pair<double,double>
764Specbnd<T>::_hertzrohn_bound
765( const unsigned N, const T*D2F )
766{
767 struct ndx{
768 static unsigned cw( const unsigned i, const unsigned j, const unsigned n ) { return i+j*n; }
769 static unsigned rw( const unsigned i, const unsigned j, const unsigned n ) { return i*n+j; }
770 };
771
772 // Form matrix S^{(N)} recursively
773 unsigned int col_S = 2;
774 short *S = new short[col_S];
775 S[0] = 1; S[1] = -1;
776#ifdef MC__SPECBND_DEBUG_HESSBND
777 mc::display( 1, col_S, S, 1, "\nMatrix S1", std::cout );
778#endif
779 for( unsigned int k=1; k<N; k++, col_S*=2 ){
780 short *Sprev = new short[k*col_S];
781 for( unsigned int i=0; i<k; i++ )
782 for( unsigned int j=0; j<col_S; j++ )
783 Sprev[j*k+i] = S[j*k+i];
784 delete[] S; S = new short[(k+1)*2*col_S];
785 for( unsigned int i=0; i<k; i++ ){
786 for( unsigned int j=0; j<col_S; j++ )
787 S[j*(k+1)+i] = S[(col_S+j)*(k+1)+i] = Sprev[j*k+i];
788 }
789 for( unsigned int j=0; j<col_S; j++ ){
790 S[j*(k+1)+k] = 1; S[(col_S+j)*(k+1)+k] = -1;
791 }
792 delete[] Sprev;
793#ifdef MC__SPECBND_DEBUG_HESSBND
794 mc::display( k+1, 2*col_S, S, k+1, "\nMatrix Sk", std::cout );
795#endif
796 }
797
798#ifdef MC__SPECBND_DEBUG_HESSBND
799 T *H = new T[N*N];
800 for( unsigned int j=0; j<N; j++ )
801 for( unsigned int i=j; i<N; i++ ){
802 if( i == j ){
803 H[i*N+i] = D2F[ndx::rw(i,i,N)];
804 continue;
805 }
806 T D2Fij;
807 if( !Op<T>::inter( D2Fij, D2F[ndx::rw(i,j,N)], D2F[ndx::rw(j,i,N)] ) )
808 D2Fij = D2F[ndx::rw(i,j,N)];
809 H[i*N+j] = H[j*N+i] = D2Fij;
810 }
811 mc::display( N, N, H, N, "\nMatrix H", std::cout );
812 delete[] H;
813#endif
814
815 // Compute lower and upper bound on spectral radius
816 std::pair<double,double> spbnd;
817 double *Lk = new double[N*N], *Uk = new double[N*N];
818 for( unsigned int k=0; k<col_S; k++ ){
819
820 for( unsigned int j=0; j<N; j++ )
821 for( unsigned int i=j; i<N; i++ ){
822 if( i == j ){
823 Lk[i*N+i] = Op<T>::l( D2F[ndx::rw(i,i,N)] );
824 Uk[i*N+i] = Op<T>::u( D2F[ndx::rw(i,i,N)] );
825 continue;
826 }
827 T D2Fij;
828 if( !Op<T>::inter( D2Fij, D2F[ndx::rw(i,j,N)], D2F[ndx::rw(j,i,N)] ) )
829 D2Fij = D2F[ndx::rw(i,j,N)];
830 Lk[i*N+j] = Lk[j*N+i] = S[k*N+i]*S[k*N+j]==1? Op<T>::l( D2Fij ): Op<T>::u( D2Fij );
831 Uk[i*N+j] = Uk[j*N+i] = S[k*N+i]*S[k*N+j]==1? Op<T>::u( D2Fij ): Op<T>::l( D2Fij );
832 }
833#ifdef MC__SPECBND_DEBUG_HESSBND
834 mc::display( N, N, Lk, N, "\nMatrix Lk", std::cout );
835 mc::display( N, N, Uk, N, "\nMatrix Uk", std::cout );
836#endif
837
838 double*DLk = mc::dsyev_wrapper( N, Lk );
839 if( !DLk ){
840 delete[] Lk; delete[] Uk; delete[] S;
841 throw typename Specbnd<T>::Exceptions( Specbnd<T>::Exceptions::HESSBND );
842 }
843#ifdef MC__SPECBND_DEBUG_HESSBND
844 mc::display( 1, N, DLk, 1, "\nMatrix DLk", std::cout );
845#endif
846 spbnd.first = k? std::min( spbnd.first, min(N,DLk) ): min(N,DLk);
847 delete[] DLk;
848
849 double*DUk = mc::dsyev_wrapper( N, Uk );
850 if( !DUk ){
851 delete[] Lk; delete[] Uk; delete[] S;
