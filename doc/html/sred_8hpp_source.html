227template <typename COMP=std::less<unsigned>>
228struct lt_CtrMon
229{
230 // Comparison operator
231 template <typename KEY>
232 bool operator
233 ()
234 ( std::pair< unsigned, SMon<KEY,COMP> const* > const& iMon,
235 std::pair< unsigned, SMon<KEY,COMP> const* > const& jMon )
236 const
237 {
238 // Order based on constraint index first
239 if( iMon.first < jMon.first ) return true;
240 if( iMon.first > jMon.first ) return false;
241 // Order based on multiplier monomial next
242 if( lt_SMon<COMP>()( *iMon.second, *jMon.second ) ) return true;
243 if( lt_SMon<COMP>()( *jMon.second, *iMon.second ) ) return false;
244 // Pairs are identical on reaching this point
245 return false;
246 }
247};

723template <typename KEY, typename COMP>
724template <typename POL >
725inline
726unsigned
727SRed<KEY,COMP>::search_reductions
728( std::set<unsigned> const& ndxSPol, POL const* pSPol, map_SPoly const& (POL::*mapmon)() const,
729 bool const LINEARIZE )
730{
731 _Reduc_reset();
732 if( ndxSPol.empty() ) return 0;
733#ifdef MC__SRED_DEBUG
734 for( auto const& i : ndxSPol )
735 std::cout << "pSPol[" << i << "]: " << pSPol[i] << std::endl;
736#endif
737
738 // Keep track of participating variables in reduction problem
739 _VarDeg.expr.clear();
740 _VarDeg.tord = 0;
741 for( auto const& mon : _SetMon )
742 _VarDeg.hull( mon );
743 for( auto const& i : ndxSPol )
744 for( auto const& [mon,coef] : (pSPol[i].*mapmon)() ){
745 if( coef == 0. ) continue;
746 _VarDeg.hull( mon );
747 }
748 _SetVar.clear();
749 for( auto const& [var,ord] : _VarDeg.expr )
750 _SetVar.insert( var );
751
752 // Generate set of multiplier/divider monomials
753 _sizemon();
754
755 // Create sets of new monomials for each polynomial constraint and each multiplier/divider monomial
756 for( t_size k=0; k<_nmon; ++k ){
757 // Create candidate multiplier/divider monomial
758 t_SMon kmon;
759 unsigned j = 0;
760 for( auto const& var : _SetVar ){
761 unsigned const& exp = _expmon[ k*_nvar+(j++) ];
762 if( !exp ) continue;
763 kmon.expr[var] = exp;
764 kmon.tord += exp;
765 }
766 if( !kmon.tord ) continue;
767 auto [ktmon,ins] = _SetMonOp.insert( kmon );
768#ifdef MC__SRED_DEBUG
769 std::cout << "kmon: " << kmon.display(0) << std::endl;
770#endif
771 // Test if multiplier monomial exists in product with monomials in constraints
772 for( auto const& i : ndxSPol ){
773 for( auto const& [mon,coef] : (pSPol[i].*mapmon)() ){
774 if( coef == 0. ) continue;
775 auto jtmon = (mon.tord>1? _SetMon.find( mon ): _SetMon.begin());
776 if( jtmon == _SetMon.end() ) continue;
777 // New multiplier term added if not present and has order greater than one
778 t_SMon monnew( kmon );
779 if( mon.tord ) monnew += mon;
780 if( monnew.tord > 1 && !_SetMon.count( monnew ) ){
781 auto [iktmon,ins] = _SetMonNew.insert( monnew );
782 _NewMul[std::make_pair(i,&*ktmon)].insert( &*iktmon );
783#ifdef MC__SRED_DEBUG
784 std::cout << " ctr " << i << ", mon " << jtmon->display(0) << std::endl;
785#endif
786 }
787 if( options.NODIV ) continue;
788 // New division term added if not present and has order greater than one
789 if( !kmon.subseteq( mon ) || ( mon.tord > kmon.tord+1 && !_SetMon.count( mon - kmon ) ) ){
790#ifdef MC__SRED_DEBUG
791 std::cout << " imon " << mon.display(0) << " - kmon " << kmon.display(0) << std::endl;
792#endif
793 t_SMonExt monnew( mon ); monnew -= kmon;
794 auto [iktmon,ins] = _SetMonNew.insert( monnew );
795 _NewDiv[std::make_pair(i,&*ktmon)].insert( &*iktmon );
796 }
797 }
798 }
799 }
800
801 // Search reduction constraints
802#if defined(MC__USE_GUROBI)
803 _MIP_optimize( ndxSPol );
804#else
805 throw Exceptions( Exceptions::MIPERR );
806#endif
807
808 // Coefficient matrix and right-hand-side vector holding the reduction constraints
809 for( auto const& [keyctr,grbvar] : _MIP_CtrMul ){
810 if( grbvar.get(GRB_DoubleAttr_X) < 0.9 ) continue;
811 auto const& [ctr,monop] = keyctr;
812 _ndxCtrUse.insert( ctr );
813 _rhs.push_back( 0. );
814 _coef.push_back( std::map<t_SMonExt const*,double,lt_pSMonExt<COMP>>() );
815 for( auto const& [mon,coef] : (pSPol[ctr].*mapmon)() ){
816 if( coef == 0. ) continue;
817 t_SMonExt monnew( mon ); monnew += *monop;
818 auto itmonnew = _SetMonNew.find( monnew );
819 if( itmonnew == _SetMonNew.end() )
820 _rhs.back() += std::make_pair( mon + *monop, coef );
821 else{
822#ifdef MC__SRED_CHECK
823 auto itmip = _MIP_NewMul.find( &monnew );
824#ifdef MC__SRED_DEBUG
825 std::cout << "ctr:" << ctr << " monop:" << monop->display(0) << " monnew:" << monnew.display(0)
826 << " " << itmip->second.get(GRB_DoubleAttr_X) << std::endl;
827#endif
828 assert( itmip != _MIP_NewMul.end() && itmip->second.get(GRB_DoubleAttr_X) > 0.9 );
829#endif
830 _new.insert( std::make_pair( &*itmonnew, _new.size() ) );
831 _coef.back().insert( std::make_pair( &*itmonnew, coef ) );
832 }
833 }
834 }
835
836 for( auto const& [keyctr,grbvar] : _MIP_CtrDiv ){
837 if( grbvar.get(GRB_DoubleAttr_X) < 0.9 ) continue;
838 auto const& [ctr,monop] = keyctr;
839 _ndxCtrUse.insert( ctr );
840 _rhs.push_back( 0. );
841 _coef.push_back( std::map<t_SMonExt const*,double,lt_pSMonExt<COMP>>() );
842 for( auto const& [mon,coef] : (pSPol[ctr].*mapmon)() ){
843 if( coef == 0. ) continue;
844 t_SMonExt monnew( mon ); monnew -= *monop;
845 auto itmonnew = _SetMonNew.find( monnew );
846 if( itmonnew == _SetMonNew.end() )
847 _rhs.back() += std::make_pair( mon - *monop, coef );
848 else{
849#ifdef MC__SRED_CHECK
850 auto itmip = _MIP_NewDiv.find( &monnew );
851 assert( itmip != _MIP_NewDiv.end() && itmip->second.get(GRB_DoubleAttr_X) > 0.9 );
852#endif
853 _new.insert( std::make_pair( &*itmonnew, _new.size() ) );
854 _coef.back().insert( std::make_pair( &*itmonnew, coef ) );
855 }
856 }
857 }
858
859#ifdef MC__SRED_DEBUG
860 unsigned irhs = 0;
861 for( auto const& pol : _rhs )
862 std::cout << "_rhs[" << irhs++ << "]: " << pol << std::endl;
863 for( auto const& [monnew,pos] : _new )
864 std::cout << "_new[" << monnew->display(0) << "]: " << pos << std::endl;
865#endif
866
867 // Perform LU decomposition of coefficient matrix
868 _mat.resize( _rhs.size(), _new.size() );
869 _mat.zero();
870 for( unsigned irow=0; irow<_coef.size(); ++irow )
871 for( auto const& [monnew,coef] : _coef[irow] )
872 _mat( irow, _new[monnew] ) = coef;
873
874 CPPL::dgematrix L, U;
875 std::vector<int> PIV;
876 int INFO = CPPL::dgetrf( _mat, U, L, PIV );
877 if( INFO < 0 ) throw Exceptions( Exceptions::LUERR );
878#ifdef MC__SRED_DEBUG
879 std::cout << "\nLU DECOMPOSITION OF COEFFICIENT MATRIX: "
880 << INFO << std::endl
881 << "\nMatrix U:\n" << U
882 << "\nMatrix L:\n" << L
883 << "\nVector PIV:\n";
884 for( auto&& ip : PIV ) std::cout << ip << std::endl;
885#endif
886
887 // Recover reduction constraints
888 for( unsigned irow=0; irow<_new.size(); ++irow ){
889 for( unsigned krow=0; krow<irow; ++krow )
890 _rhs[PIV[irow]] -= L(irow,krow) * _rhs[PIV[krow]];
891 _rhs[PIV[irow]] /= L(irow,irow);
892 }
893 for( unsigned irow=_new.size(); irow<_rhs.size(); ++irow ){
894 for( unsigned krow=0; krow<_new.size(); ++krow )
895 _rhs[PIV[irow]] -= L(irow,krow) * _rhs[PIV[krow]];
896 }
897 _RedCtr.clear();
898 for( unsigned irow=0; irow<_rhs.size()-_new.size(); ++irow ){
899 _rhs[PIV[_new.size()+irow]].clean( options.ZEROTOL );
900 if( _rhs[PIV[_new.size()+irow]].mapmon().empty() ) continue;
901 _RedCtr.push_back( _rhs[PIV[_new.size()+irow]] );
902 }
903
904 // Search linearization variables
905 return( LINEARIZE? _search_linearizations( ndxSPol, pSPol, mapmon ): _RedCtr.size() );
906}

912SRed<KEY,COMP>::_search_linearizations
913( std::set<unsigned> const& ndxSPol, POL const* pSPol, map_SPoly const& (POL::*mapmon)() const )
914{
915 _Elim_reset();
916 //if( _ndxCtrUse.empty() && _RedCtr.empty() ) return 0;
917 if( _RedCtr.empty() ) return 0;
918
919 // Coefficient matrix and right-hand-side vector holding the linearized constraints
920 // Process original polynomial constraints
921 unsigned nvar = 0, nrhs = 0;
922/*
923 for( auto const& ictr : ndxSPol ){
924 //for( auto const& ictr : _ndxCtrUse ){
925 if( (pSPol[ictr].*mapmon)().rbegin()->first.tord < 2 ) continue;
926 nrhs++;
927 _lhs.push_back( std::map<t_SMon const*,double,lt_pSMon<COMP>>() );
928 for( auto const& [mon,coef] : (pSPol[ictr].*mapmon)() ){
929 if( coef == 0. || mon.tord < 2 )
930 continue; // exclude zero, constant and linear terms from elimination
931 auto const& [itmon,ins] = _mapMonVar.insert( std::make_pair( &mon, nvar ) );
932 if( ins ){
933 _mapVarMon.insert( std::make_pair( nvar, &mon ) );
934 _mapVarWeight[nvar] = 1;
935 nvar++;
936 }
937 else{
938 _mapVarWeight[itmon->second]++;
939 }
940 _lhs.back().insert( std::make_pair( itmon->first, coef ) );
941 }
942 }
943*/
944 // Process reduction polynomial constraints
945 for( unsigned ired=0; ired<_RedCtr.size(); ++ired ){
946 if( _RedCtr[ired].mapmon().rbegin()->first.tord < 2 ) continue;
947 nrhs++;
948 _lhs.push_back( std::map<t_SMon const*,double,lt_pSMon<COMP>>() );
949 for( auto const& [mon,coef] : _RedCtr[ired].mapmon() ){
950 if( coef == 0. || mon.tord < 2 )
951 continue; // exclude zero, constant and linear terms from elimination
952 auto const& [itmon,ins] = _mapMonVar.insert( std::make_pair( &mon, nvar ) );
953 if( ins ){
954 _mapVarMon.insert( std::make_pair( nvar, &mon ) );
955 _mapVarWeight[nvar] = 1;
956 nvar++;
957 }
958 else{
959 _mapVarWeight[itmon->second]++;
960 }
961 _lhs.back().insert( std::make_pair( itmon->first, coef ) );
962 }
963 }
964#ifdef MC__SRED_DEBUG
965 for( auto const& [mon,ndx] : _mapMonVar )
966 std::cout << "_mapMonVar[" << mon->display(0) << "]: " << ndx << std::endl;
967#endif
968
969 // Terminate if size zero
970 if( !nrhs || !nvar ) return 0;
971
972 // Map of variable weights
973 for( auto const& [var,wei] : _mapVarWeight ){
974 _mapWeightVar.insert( std::make_pair( wei, var ) );
975#ifdef MC__SRED_DEBUG
976 std::cout << "_mapWeightVar[" << var << "]: " << wei << std::endl;
977#endif
978 }
979
980 // Perform LU decomposition of coefficient matrix
981 _mat.resize( nrhs, nvar );
982 _mat.zero();
983 for( unsigned irow=0; irow<_lhs.size(); ++irow )
984 for( auto const& [mon,coef] : _lhs[irow] )
985 _mat( irow, _mapMonVar[mon] ) = coef;
986#ifdef MC__SRED_DEBUG
987 std::cout << "\nMatrix A = \n" << _mat;
988#endif
989
990 CPPL::dgematrix cpmat( _mat );
991 CPPL::dcovector S;
992 CPPL::dgematrix U, VT;
993 int INFO = cpmat.dgesvd( S, U, VT );
994 if( INFO < 0 ) throw Exceptions( Exceptions::SVDERR );
995#ifdef MC__SRED_DEBUG
996 std::cout << "\nSV DECOMPOSITION OF COEFFICIENT MATRIX: "
997 << INFO << std::endl
998 << "\nMatrix U:\n" << U
999 << "\nVector S:\n" << S
1000 << "\nMatrix VT:\n" << VT;
1001#endif
1002
1003 unsigned rkmat = _rank( S.l, S.array );
1004#ifdef MC__SRED_DEBUG
1005 std::cout << "\n Rank: " << rkmat << std::endl;
1006#endif
1007
1008 // Can all monomials be eliminated?
1009 if( rkmat == _mapMonVar.size() ){
1010 for( auto const& [mon,ndx] : _mapMonVar )
1011 _setMonElim.insert( *mon );
1012 return rkmat;
1013 }
1014
1015 // Else identify maximal elimination
1016 CPPL::dgematrix U1T( rkmat, nrhs );
1017 for( unsigned i=0; i<nrhs; ++i )
1018 for( unsigned j=0; j<rkmat; ++j )
1019 U1T(j,i) = U(i,j);
1020 CPPL::dgematrix U1TA = U1T * _mat;
1021#ifdef MC__SRED_DEBUG
1022 std::cout << "\nMatrix U1T * A = \n" << U1TA;
1023 CPPL::dgematrix Smat( nrhs, nvar );
1024 Smat.zero();
1025 for( unsigned i=0; i<rkmat; ++i )
1026 Smat(i,i) = S(i);
1027 std::cout << "\nMatrix S * VT = \n" << Smat * VT;
1028#endif
1029
1030 CPPL::dgematrix SVTbasis( rkmat, rkmat );
1031 CPPL::dgbmatrix Sband;
1032 SVTbasis.zero();
1033 unsigned irk = 0;
1034 for( auto itwei=_mapWeightVar.crbegin(); irk<rkmat && itwei!=_mapWeightVar.crend(); ++itwei ){
1035 auto const& [wei,var] = *itwei;
1036#ifdef MC__SRED_DEBUG
1037 std::cout << "var = " << var << " wei = " << wei << " irk = " << irk << std::endl;
1038#endif
1039 for( unsigned i=0; i<rkmat; ++i )
1040 SVTbasis(i,irk) = U1TA(i,var);
1041 if( !irk ){
1042 _setMonElim.insert( *_mapVarMon[var] );
1043 irk++;
1044 continue;
1045 }
1046 CPPL::dgematrix cpbasis( SVTbasis );
1047 cpbasis.dgesvd( Sband );
1048 if( _rank( Sband.m, Sband.array ) == irk+1 ){
1049 _setMonElim.insert( *_mapVarMon[var] );
1050 irk++;
1051 continue;
1052 }
1053 }
1054
1055 return rkmat;
1056}

1329// Based on a code by John Burkardt
1330// http://people.sc.fsu.edu/~jburkardt/cpp_src/monomial/monomial.html
1331template <typename KEY, typename COMP>
1332inline
1333void
1334SRed<KEY,COMP>::_next_expmon
1335( unsigned *x, const unsigned m )
1336const
1337{
1338 // Ensure that M>=1
1339 assert( m );
1340
1341 // Find I, the index of the rightmost nonzero entry of X.
1342 unsigned i=0, im1, t;
1343 for( unsigned j=m; j>=1; j-- ){
1344 if ( x[j-1] ){
1345 i = j;
1346 break;
1347 }
1348 }
1349
1350 // set T = X(I)
1351 // set X(I) to zero,
1352 // increase X(I-1) by 1,
1353 // increment X(D) by T-1.
1354 if( !i ){
1355 x[m-1] = 1;
1356 return;
1357 }
1358 else if( i == 1 ){
1359 t = x[0] + 1;
1360 im1 = m;
1361 }
1362 else{
1363 t = x[i-1];
1364 im1 = i - 1;
1365 }
1366
1367 x[i-1] = 0;
1368 x[im1-1] = x[im1-1] + 1;
1369 x[m-1] = x[m-1] + t - 1;
1370}

1375SRed<KEY,COMP>::_loc_expmon
1376( const unsigned *iexp )
1377const
1378{
1379 unsigned ord = 0;
1380 for( unsigned i=0; i<_nvar; i++ ) ord += iexp[i];
1381#ifdef MC__SRED_CHECK
1382 assert( ord<_nord+2 );
1383#endif
1384 t_size pos = _posord[ord];

1387 for( unsigned i=0; i<_nvar-1; i++ ){
1388 p--;
1389 for( unsigned j=0; j<iexp[i]; j++ )
1390 pos += _get_binom( p-1+ord-j, ord-j );
1391 ord -= iexp[i];
1392 }
