416 class Exceptions
417 {
418 public:
420 enum TYPE{
421 DIV=1,
422 INV,
423 LOG,
424 SQRT,
425 DPOW,
426 TAN,
427 ACOS,
428 ASIN,
429 COMPOSE,
430 INIT=-1,
431 INCON=-2,
432 MODEL=-3,
433 INTERNAL = -4,
434 UNDEF=-33
435 };
436
437 Exceptions( TYPE ierr ) : _ierr( ierr ){}
439 int ierr(){ return _ierr; }
441 std::string what(){
442 switch( _ierr ){
443 case DIV:
444 return "mc::SCModel\t Division by zero scalar";
445 case INV:
446 return "mc::SCModel\t Inverse operation with zero in range";
447 case LOG:
448 return "mc::SCModel\t Log operation with non-positive numbers in range";
449 case SQRT:
450 return "mc::SCModel\t Square-root operation with negative numbers in range";
451 case DPOW:
452 return "mc::SCModel\t Real power operation with negative numbers in range";
453 case TAN:
454 return "mc::SCModel\t Tangent operation with (k+1/2)·PI in range";
455 case ACOS:
456 return "mc::SCModel\t Cosine inverse operation with range outside [-1,1]";
457 case ASIN:
458 return "mc::SCModel\t Sine inverse operation with range outside [-1,1]";
459 case COMPOSE:
460 return "mc::SCModel\t Chebyshev composition failed";
461 case INIT:
462 return "mc::SCModel\t Chebyshev variable initialization failed";
463 case INCON:
464 return "mc::SCModel\t Inconsistent bounds with template parameter arithmetic";
465 case MODEL:
466 return "mc::SCModel\t Operation between variables in different model environments";
467 case UNDEF:
468 return "mc::SCModel\t Feature not yet implemented";
469 case INTERNAL:
470 default:
471 return "mc::SCModel\t Internal error";
472 }
473 }
474
475 private:
476 TYPE _ierr;
477 };

1486template <typename T, typename KEY, typename COMP>
1487template <typename VEC>
1488inline
1489VEC
1490SCModel<T,KEY,COMP>::_to_chebyshev
1491( VEC const& veccoef )
1492const
1493{
1494 // Converts COEFF in monomial basis to the Chebyshev basis.
1495 VEC vecnew = veccoef;
1496 int const N = vecnew.size()-1;
1497 if( N <= 1 ) return vecnew;
1498 // TP = .5D0**(N-1)
1499 // COEFF(N) = TP * COEFF(N)
1500 // COEFF(N-1) = TP * COEFF(N-1)
1501 double TP = std::pow(0.5,N-1);
1502 vecnew[N] *= TP;
1503 vecnew[N-1] *= TP;
1504 // do 20 J = N-2, 0, -1
1505 // TP = 2.D0 * TP
1506 // COEFF(J) = TP * COEFF(J)
1507 // COEFF(J+1) = 2.D0 * COEFF(J+1)
1508 for( int J=N-2; J>=0; --J ){
1509 TP *= 2;
1510 vecnew[J] *= TP;
1511 vecnew[J+1] *= 2;
1512 // do 10 I = J, N-2
1513 // COEFF(I) = COEFF(I) + COEFF(I+2)
1514 for( int I=J; I<N-1; ++I )
1515 vecnew[I] += vecnew[I+2];
1516 }
1517 return vecnew;
1518}

1575template <typename T, typename KEY, typename COMP>
1576template <typename PUNIV>
1577inline
1578void
1579SCModel<T,KEY,COMP>::_chebinterp
1580( PUNIV const& f, unsigned const maxord ) // range of f assumed as [-1,1]
1581{
1582 _xinterp.resize( maxord+1 );
1583 _finterp.resize( maxord+1 );
1584 double mulconst( PI/(2.*double(maxord+1)) );
1585 for( unsigned i=0; i<=maxord; i++ ){
1586 _xinterp[i] = std::cos( mulconst*(2.*double(i)+1.) );
1587 _finterp[i] = f( _xinterp[i] );
1588 }
1589
1590 _coefuniv.resize( maxord+1 );
1591 switch( maxord ){
1592 case 0:
1593 _coefuniv[0] = _finterp[0];
1594 return;
1595 case 1:
1596 _coefuniv[0] = 0.5 * ( _finterp[0] + _finterp[1] );
1597 _coefuniv[1] = ( _finterp[1] - _finterp[0] ) / ( _xinterp[1] - _xinterp[0] );
1598 return;
1599 default:
1600 for( unsigned i=0; i<=maxord; i++ ){
1601 double mulconst2( std::cos( mulconst*double(i) ) ),
1602 mulconst3( 4*mulconst2*mulconst2-2 ),
1603 b0( 0 ), b1( 0 );
1604 b0 = _finterp[maxord];
1605 b1 = _finterp[maxord-1] + mulconst3*b0;
1606 for( unsigned j=maxord-2; j>1; j-=2 ){
1607 b0 = _finterp[j] + mulconst3*b1 - b0;
1608 b1 = _finterp[j-1] + mulconst3*b0 - b1;
1609 }
1610 if( !(maxord%2) )
1611 b0 = _finterp[0] + mulconst3*b1 - b0 - b1;
1612 else{
1613 b0 = _finterp[1] + mulconst3*b1 - b0;
1614 b0 = _finterp[0] + mulconst3*b0 - b1 - b0;
1615 }
1616 _coefuniv[i] = 2./double(maxord+1)*mulconst2*b0;
1617 }
1618 _coefuniv[0] *= 0.5;
1619 return;
1620 }
1621}

1984template <typename T, typename KEY, typename COMP>
1985inline
1986void
1987SCModel<T,KEY,COMP>::_sprod1D
1988( std::map<unsigned,t_poly> const& sp1map,
1989 std::map<unsigned,t_poly> const& sp2map,
1990 t_poly& coefmon, T& rem, t_var const& ndxvar,
1991 typename t_var::const_iterator itvar )
1992const
1993{
1994 // construct product matrix of polynomial coefficients
1995 auto itvarnext = itvar;
1996 if( !ndxvar.empty() ) std::advance( itvarnext, 1 );
1997 std::map<std::pair<unsigned,unsigned>,t_poly> sp12map;
1998 for( auto& [ndx1,coefmon1] : sp1map ){
1999 // empty monomial in sp1
2000 if( coefmon1.empty() ) continue;
2001 // constant monomial in sp1
2002 if( coefmon1.size() == 1 && !coefmon1.begin()->first.tord ){
2003 for( auto& [ndx2,coefmon2] : sp2map ){
2004 auto [it12,ins] = sp12map.insert( std::make_pair( std::make_pair(ndx1,ndx2), t_poly() ) );
2005 assert( ins ); // map is initially empty
2006 _sscal1D( coefmon2, coefmon1.begin()->second, it12->second );
2007 }
2008 continue;
2009 }
2010 // general monomial in sp1
2011 for( auto& [ndx2,coefmon2] : sp2map ){
2012 // empty monomial in sp2
2013 if( coefmon2.empty() ) continue; // no term
2014 // constant monomial in sp2
2015 if( coefmon2.size() == 1 && !coefmon2.begin()->first.tord ){
2016 auto [it12,ins] = sp12map.insert( std::make_pair( std::make_pair(ndx1,ndx2), t_poly() ) );
2017 assert( ins ); // map is initially empty
2018 _sscal1D( coefmon1, coefmon2.begin()->second, it12->second );
2019 continue;
2020 }
2021#ifdef MC__SCMODEL_DEBUG_SPROD
2022 std::cout << "Term (" << ndx1 << "," << ndx2 << "):\n";
2023#endif
2024 std::map<unsigned,t_poly> sp11map, sp22map;
2025 for( auto& mon1 : coefmon1 )
2026 _svec1D( itvarnext, mon1, sp11map );
2027#ifdef MC__SCMODEL_DEBUG_SPROD
2028 _sdisp1D( sp11map, itvarnext, "Poly #1: " );
2029#endif
2030 for( auto& mon2 : coefmon2 )
2031 _svec1D( itvarnext, mon2, sp22map );
2032#ifdef MC__SCMODEL_DEBUG_SPROD
2033 _sdisp1D( sp22map, itvarnext, "Poly #2: " );
2034#endif
2035 auto [it12,ins] = sp12map.insert( std::make_pair( std::make_pair(ndx1,ndx2), t_poly() ) );
2036 assert( ins ); // map is initially empty
2037 _sprod1D( sp11map, sp22map, it12->second, rem, ndxvar, itvarnext );
2038 }
2039 }
2040
2041 // construct 1D product result and augment remainder as appropriate
2042 coefmon.clear();
2043 for( auto const& [ndx12,coefmon12] : sp12map ){
2044 auto const& [ndx1,ndx2] = ndx12;
2045 // Product involving two constant terms
2046 if( !ndx1 && !ndx2 )
2047 _slift1D( coefmon12, 1., coefmon );
2048 // Product involving exactly one constant term
2049 else if( !ndx1 || !ndx2 )
2050 _slift1D( coefmon12, 1., coefmon, rem, itvar, ndx1+ndx2 );
2051 // Product between non-constant terms
2052 else{
2053 switch( options.BASIS ){
2054 // Chebyshev basis functions
2055 case Options::CHEB:
2056 if( ndx1+ndx2 <= _maxord )
2057 _slift1D( coefmon12, .5, coefmon, rem, itvar, ndx1+ndx2 );
2058 else
2059 _slift1D( coefmon12, .5, rem, ndx1+ndx2 );
2060 if( ndx1 == ndx2 )
2061 _slift1D( coefmon12, .5, coefmon );
2062 else if( ndx1 > ndx2 )
2063 _slift1D( coefmon12, .5, coefmon, rem, itvar, ndx1-ndx2 );
2064 else
2065 _slift1D( coefmon12, .5, coefmon, rem, itvar, ndx2-ndx1 );
2066 break;
2067 // Power basis functions
2068 case Options::MONOM:
2069 if( ndx1+ndx2 <= _maxord )
2070 _slift1D( coefmon12, 1., coefmon, rem, itvar, ndx1+ndx2 );
2071 else
2072 _slift1D( coefmon12, 1., rem, ndx1+ndx2 );
2073 break;
2074 }
2075 }
2076 }
2077#ifdef MC__SCMODEL_DEBUG_SPROD
2078 _sdisp1D( coefmon, "Prod: " );
2079 std::endl;
2080#endif
2081}

3005template <typename T, typename KEY, typename COMP>
3006inline
3007void
3008SCVar<T,KEY,COMP>::_to_monomial
3009( KEY const& id, typename SCVar<T,KEY,COMP>::t_poly& coefmon )
3010const
3011{
3012 // Get coefficients in univariate polynomial representation w.r.t variable id
3013 std::vector<t_poly> veccoef( nord()+1, t_poly() );
3014 for( auto [mon,coef] : coefmon ){
3015 auto ie = mon.expr.find( id );
3016 if( ie == mon.expr.end() ) // no dependence on variable id
3017 veccoef[ 0 ].insert( std::make_pair( mon, coef ) );
3018 else{
3019 auto const& iord = ie->second;
3020 t_mon monmod( mon.tord - iord, mon.expr );
3021 monmod.expr.erase( id ); // remove T[id] entry
3022 veccoef[ iord ].insert( std::make_pair( monmod, coef ) );
3023 }
3024 }
3025
3026 // Nothing to convert if independent of current variable #i
3027 bool nodep = true;
3028 for( unsigned k=1; nodep && k<=nord(); k++ )
3029 if( !veccoef[k].empty() ) nodep = false;
3030 if( nodep ) return;
3031
3032 // Convert current variable #i to monomial
3033 // Based on SCONCM function (http://www.netlib.org/math/docpdf/ch11-03.pdf)
3034 double TP = 1e0;
3035 // do 20 J = 0, N-2
3036 for( int j=0; j<(int)nord()-1; j++ ){
3037 // do 10 I = N-2, J, -1
3038 for( int i=(int)nord()-2; i>=j; i-- ){
3039 // COEFF(I) = COEFF(I) - COEFF(I+2)
3040 for( auto& [mon,coef] : veccoef[i+2] ){
3041 auto [itmon, ins] = veccoef[i].insert( std::make_pair( mon, -coef ) );
3042 if( !ins ) itmon->second -= coef;
3043 }
3044 // 10 continue
3045 }
3046 // COEFF(J+1) = .5E0 * COEFF(J+1)
3047 for( auto& [mon,coef] : veccoef[j+1] ) coef /= 2e0;
3048 // COEFF(J) = TP * COEFF(J)
3049 for( auto& [mon,coef] : veccoef[j] ) coef *= TP;
3050 // TP = 2.E0 * TP
3051 TP *= 2e0;
3052 // 20 continue
3053 }
3054 // COEFF(N) = TP * COEFF(N)
3055 for( auto& [mon,coef] : veccoef[nord()] ) coef *= TP;
3056 // COEFF(N-1) = TP * COEFF(N-1)
3057 for( auto& [mon,coef] : veccoef[nord()-1] ) coef *= TP;
3058
3059 // Collect monomials back together into <a>coefmon</a>
3060 coefmon = veccoef[0];
3061 for( unsigned iord=1; iord<=nord(); iord++ ){
3062 for( auto const& [mon,coef] : veccoef[iord] ){
3063 t_mon monmod( mon.tord + iord, mon.expr );
3064 monmod.expr[ id ] = iord;
3065 coefmon[ monmod ] = coef;
3066 }
3067 }
3068}
