153 class Exceptions
154 {
155 public:
157 enum TYPE{
158 SUB=1,
159 DIV,
160 CONV
161 };
162
163 Exceptions( TYPE ierr ) : _ierr( ierr ){}
165 int ierr(){ return _ierr; }
167 std::string what(){
168 switch( _ierr ){
169 case SUB:
170 return "mc::SMon\t Subtraction of a monomial that is not a proper subset";
171 case DIV:
172 return "mc::SMon\t Division by a factor greater than the greatest common exponent";
173 case CONV:
174 return "mc::SMon\t Conversion with different variable indexing failed";
175 }
176 return "mc::SMon\t Undocumented error";
177 }
178 private:
179 TYPE _ierr;
180 };

193template <typename KEY, typename COMP>
194inline std::string
195SMon<KEY,COMP>::display
196( int const& basis )
197const
198{
199 std::ostringstream out;
200 // Sparse monomial
201 if( expr.empty() ) out << "1";
202 for( auto ie=expr.begin(); ie!=expr.end(); ++ie ){
203 if( ie != expr.begin() ) out << "·";
204 switch( basis ){
205 case 0:
206 out << "[" << *ptr(ie->first) << "]";
207 if( ie->second > 1 ) out << "^" << ie->second;
208 break;
209 default:
210 out << "T" << ie->second << "[" << *ptr(ie->first) << "]";
211 break;
212 }
213 }
214 return out.str();
215}

228template <typename KEY, typename COMP>
229inline bool
230SMon<KEY,COMP>::operator==
231( SMon<KEY,COMP> const& Mon )
232const
233{
234 // Order monomials based on their total order first
235 if( tord < Mon.tord
236 || tord > Mon.tord ) return false;
237 // Account for the case of an empty list
238 if( !tord ) return true;
239 // Order in graded lexicographic order next
240 for( auto it1=expr.begin(), it2=Mon.expr.begin(); it1!=expr.end(); ++it1, ++it2 )
241 if( COMP()( it1->first, it2->first )
242 || COMP()( it2->first, it1->first )
243 || it1->second != it2->second ) return false;
244 return true;
245}

265template <typename KEY, typename COMP>
266inline unsigned int
267SMon<KEY,COMP>::gcexp
268()
269const
270{
271 if( expr.size() <= 1 ) return tord;
272 auto&& it = expr.cbegin();
273 unsigned gce = it->second;
274 for( ++it; gce>1 && it!=expr.cend(); ++it )
275 gce = std::gcd( it->second, gce );
276 return gce;

324const
325{
326 for( auto&& it1=expr.begin(); it1!=expr.end(); ++it1 )
327 // Monomials share a common variable (regardless of exponent)
328 if( Mon.expr.find( it1->first ) != Mon.expr.end() ) return true;

332template <typename KEY, typename COMP>
333inline bool
334SMon<KEY,COMP>::subset
335( SMon<KEY,COMP> const& Mon )
336const
337{
338 // Total order must be larger for Mon than *this
339 if( tord >= Mon.tord ) return false;
340 for( auto&& it1=expr.begin(); it1!=expr.end(); ++it1 ){
341 auto&& it2 = Mon.expr.find( it1->first );
342 // Mon must comprise (at least) the same variables as *this, and the order
343 // of each variable in Mon must have a larger order than in *this
344 if( it2 == Mon.expr.end() || it1->second > it2->second ) return false;
345 }
346 return true;
347}

353const
354{
355 // Total order must be larger or equal for Mon than *this
356 if( tord > Mon.tord ) return false;
357 for( auto&& it1=expr.begin(); it1!=expr.end(); ++it1 ){
358 auto&& it2 = Mon.expr.find( it1->first );
359 // Mon must comprise (at least) the same variables as *this, and the order
360 // of each variable in Mon must have a larger order than in *this
361 if( it2 == Mon.expr.end() || it1->second > it2->second ) return false;
362 }
363 return true;

366template <typename KEY, typename COMP>
367inline SMon<KEY,COMP>&
368SMon<KEY,COMP>::operator*=
369( unsigned const& factor )
370{
371 // Return if factor is unity
372 if( factor == 1 )
373 return *this;
374
375 // multiply monomial partial orders by factor
376 tord = 0;
377 for( auto& varpow : expr ){
378 varpow.second *= factor;
379 tord += varpow.second;
380 }
381 return *this;
382}

393template <typename KEY, typename COMP>
394inline SMon<KEY,COMP>&
395SMon<KEY,COMP>::operator/=
396( unsigned const& factor )
397{
398 // Return if factor is unity
399 if( factor == 1 )
400 return *this;
401
402 // factor may not be greater than the least exponent
403 if( lexp() < factor )
404 throw typename SMon<KEY,COMP>::Exceptions( SMon<KEY,COMP>::Exceptions::DIV );
405
406 // divide monomial partial orders by factor
407 tord = 0;
408 for( auto& varpow : expr ){
409 varpow.second /= factor;
410 tord += varpow.second;
411 }
412 return *this;
413}

426SMon<KEY,COMP>::operator+=
427( SMon<KEY,COMP> const& mon )
428{
429 for( auto const& varpow : mon.expr ){
430 auto&& it = expr.insert( varpow );
431 // If element from mon was not inserted, increment existing variable order
432 if( !it.second ) it.first->second += varpow.second;

448template <typename KEY, typename COMP>
449inline SMon<KEY,COMP>&
450SMon<KEY,COMP>::operator-=
451( SMon<KEY,COMP> const& mon )
452{
453 // mon must be a proper subset
454 if( !mon.subseteq( *this ) )
455 throw Exceptions( Exceptions::SUB );
456
457 // Return constant monomial if mon is identical
458 if( mon.tord == tord ){
459 expr.clear();
460 tord = 0;
461 return *this;
462 }
463
464 // Cancel the common terms with mon
465 for( auto& varpow : mon.expr ){
466 auto&& it = expr.find( varpow.first );
467 assert( it != mon.expr.end() );
468 if( it->second == varpow.second )
469 expr.erase( it );
470 else
471 it->second -= varpow.second;
472 }
473 tord -= mon.tord;
474 return *this;
475}

488template <typename COMP=std::less<unsigned>>
489struct lt_SMon
490{
491 template <typename KEY>
492 bool operator()
493 ( SMon<KEY,COMP> const& Mon1, SMon<KEY,COMP> const& Mon2 )
494 const
495 {
496 // Order monomials based on their total order first
497 if( Mon1.tord < Mon2.tord ) return true;
498 if( Mon1.tord > Mon2.tord ) return false;
499 // Account for the case of an empty list
500 if( Mon2.expr.empty() ) return false;
501 if( Mon1.expr.empty() ) return true;
502 // Order in graded lexicographic order next
503 for( auto&& it1=Mon1.expr.begin(),&& it2=Mon2.expr.begin(); it1!=Mon1.expr.end(); ++it1, ++it2 ){
504 if( COMP()( it1->first, it2->first ) ) return true;
505 if( COMP()( it2->first, it1->first ) ) return false;
506 if( it1->second > it2->second ) return true;
507 if( it1->second < it2->second ) return false;
508 }
509 return false;
510 }
511};

691struct lt_SMonExt
692{
693 template <typename KEY>
694 bool operator()
695 ( SMonExt<KEY,COMP> const& Mon1, SMonExt<KEY,COMP> const& Mon2 )
696 const
697 {
698 // Order extended monomials based on number of terms first
699 if( Mon1.expr.size() < Mon2.expr.size() ) return true;
700 if( Mon1.expr.size() > Mon2.expr.size() ) return false;
701 // Account for the case of an empty list
702 if( Mon2.expr.empty() ) return false;
703 if( Mon1.expr.empty() ) return true;
704 // Order in graded lexicographic order next
705 for( auto it1=Mon1.expr.begin(), it2=Mon2.expr.begin(); it1!=Mon1.expr.end(); ++it1, ++it2 ){
706 if( COMP()( it1->first, it2->first ) ) return true;
707 if( COMP()( it2->first, it1->first ) ) return false;
708 if( it1->second > it2->second ) return true;
709 if( it1->second < it2->second ) return false;
710 }
711 return false;
712 }
