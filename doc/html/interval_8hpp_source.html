395 class Exceptions
396 {
397 public:
399 enum TYPE{
400 DIV=1,
401 INV,
402 LOG,
403 SQRT,
404 ACOS,
405 ASIN,
406 TAN,
407 CHEB
408 };
409
410 Exceptions( TYPE ierr ) : _ierr( ierr ){}
412 int ierr(){ return _ierr; }
414 std::string what(){
415 switch( _ierr ){
416 case DIV:
417 return "mc::Interval\t Division by zero";
418 case INV:
419 return "mc::Interval\t Inverse with zero in range";
420 case LOG:
421 return "mc::Interval\t Log with negative values in range";
422 case SQRT:
423 return "mc::Interval\t Square-root with nonpositive values in range";
424 case ACOS:
425 return "mc::Interval\t Inverse cosine with values outside of [-1,1] range";
426 case ASIN:
427 return "mc::Interval\t Inverse sine with values outside of [-1,1] range";
428 case TAN:
429 return "mc::Interval\t Tangent with values pi/2+k*pi in range";
430 case CHEB:
431 return "mc::Interval\t Chebyshev basis outside of [-1,1] range";
432 }
433 return "mc::Interval\t Undocumented error";
434 }
435
436 private:
437 TYPE _ierr;
438 };

836inline Interval
837pow
838( Interval const& I, const int n )
839{
840 if( n == 0 ){
841 return 1.;
842 }
843 if( n == 1 ){
844 return I;
845 }
846 if( n >= 2 && n%2 == 0 ){
847 int imid = -1;
848 return Interval( std::pow(mid(I._l,I._u,0.,imid),n),
849 std::max(std::pow(I._l,n),std::pow(I._u,n)) );
850 }
851 if ( n >= 3 ){
852 return Interval( std::pow(I._l,n), std::pow(I._u,n) );
853 }
854 return inv( pow( I, -n ) );
855}

879inline Interval
880cheb
881( Interval const& I0, const unsigned n )
882{
883 Interval I(-1.,1.);
884 if( !inter( I, I0, I ) ){
885 //if ( I._l < -1.-1e1*machprec() || I._u > 1.+1e1*machprec() ){
886 throw typename Interval::Exceptions( Interval::Exceptions::CHEB );
887 }
888 switch( n ){
889 case 0: return 1.;
890 case 1: return I;
891 case 2: return 2.*sqr(I)-1.;
892 default:{
893 int kL = n - std::ceil(n*std::acos(I._l)/mc::PI); if( kL <= 0 ) kL = 0;
894 int kU = n - std::floor(n*std::acos(I._u)/mc::PI); if( kU >= (int)n ) kU = n;
895#ifdef MC__INTERVAL_CHEB_DEBUG
896 std::cout << " kL: " << kL << " kU: " << kU;
897#endif
898 if( kU-kL <= 1 ){ // monotonic part
899 double TL = std::cos(n*std::acos(I._l));
900 double TU = std::cos(n*std::acos(I._u));
901 return( TL<=TU? Interval(TL,TU): Interval(TU,TL) );
902 }
903 else if( kU-kL == 2 ){ // single extremum in range
904 double TL = std::cos(n*std::acos(I._l));
905 double TU = std::cos(n*std::acos(I._u));
906 if( (n-kL)%2 ) return( TL<=TU? Interval(TL, 1.): Interval(TU, 1.) ); // minimum
907 else return( TL<=TU? Interval(-1.,TU): Interval(-1.,TL) ); // maximum
908 }
909 break;
910 }
911 }
912 //Interval Icheb = 2.*I*cheb(I,n-1)-cheb(I,n-2);
913 //return( inter( Icheb, Icheb, Interval(-1.,1.) )? Icheb: Interval(-1.,1.) );
914 return Interval(-1.,1.);
915}

970inline Interval
971cos
972( Interval const& I )
973{
974 const int k = std::ceil(-(1.+I._l/PI)/2.); // -pi <= xL+2*k*pi < pi
975 const double l = I._l+2.*PI*k, u = I._u+2.*PI*k;
976 if( l <= 0 ){
977 if( u <= 0 ) return Interval( std::cos(l), std::cos(u) );
978 if( u >= PI ) return Interval( -1., 1. );
979 return Interval( std::min(std::cos(l), std::cos(u)), 1. );
980 }
981 if( u <= PI ) return Interval( std::cos(u), std::cos(l) );
982 if( u >= 2.*PI ) return Interval( -1., 1. );
