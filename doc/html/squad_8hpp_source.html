540template <typename COMP=std::less<unsigned>>
541struct lt_SQuad
542{
543 // Comparison operator
544 template <typename KEY>
545 bool operator
546 ()
547 ( std::pair< SMon<KEY,COMP> const*, SMon<KEY,COMP> const* > const& pMon1,
548 std::pair< SMon<KEY,COMP> const*, SMon<KEY,COMP> const* > const& pMon2 )
549 const
550 {
551 // Order based on first monomial first
552 if( lt_SMon<COMP>()( *pMon1.first, *pMon2.first ) ) return true;
553 if( lt_SMon<COMP>()( *pMon2.first, *pMon1.first ) ) return false;
554 // Order based on second monomial next
555 if( lt_SMon<COMP>()( *pMon1.second, *pMon2.second ) ) return true;
556 if( lt_SMon<COMP>()( *pMon2.second, *pMon1.second ) ) return false;
557 // Pairs are identical on reaching this point
558 return false;
559 }
560};

1034template <typename KEY, typename COMP>
1035inline void
1036SQuad<KEY,COMP>::tighten
1037( bool const threevar )
1038{
1039 _MatPSD.clear();
1040 auto itmon1 = _SetMon.cbegin();
1041 for( ; itmon1 != _SetMon.cend(); ++itmon1 ){
1042
1043 // check square monomial *itmon1 participating
1044 if( itmon1->tord && !_find( &*itmon1, &*itmon1 ) ) continue;
1045 auto itmon2 = itmon1;
1046 for( ++itmon2; itmon2 != _SetMon.cend(); ++itmon2 ){
1047
1048 // check square monomial *itmon2 and cross-term *itmon1.*itmon2 participating
1049 if( (itmon1->tord && !_find( &*itmon1, &*itmon2 )) || !_find( &*itmon2, &*itmon2 ) ) continue;
1050
1051 // add positive-definite cuts
1052 unsigned pos = _MatPSD.size();
1053 _MatPSD.push_back( map_SQuad() );
1054 auto& mat1 = _MatPSD.back();
1055 assert( _insert( mat1, &*itmon1, &*itmon1, 1. )
1056 && _insert( mat1, &*itmon2, &*itmon2, 1. )
1057 && _insert( mat1, &*itmon1, &*itmon2, -2. ) );
1058 if( !threevar && options.BASIS == Options::MONOM
1059 && !itmon1->gcexp()%2 && !itmon2->gcexp()%2 ) continue;
1060 _MatPSD.push_back( map_SQuad() );
1061 auto& mat2 = _MatPSD.back();
1062 assert( _insert( mat2, &*itmon1, &*itmon1, 1. )
1063 && _insert( mat2, &*itmon2, &*itmon2, 1. )
1064 && _insert( mat2, &*itmon1, &*itmon2, 2. ) );
1065
1066 if( !threevar ) continue;
1067 auto itmon3 = itmon2;
1068 for( ++itmon3; itmon3 != _SetMon.cend(); ++itmon3 ){
1069
1070 // check square monomial *itmon2 and cross-term *itmon1.*itmon2 participating
1071 if( (itmon1->tord && !_find( &*itmon1, &*itmon3 )) || !_find( &*itmon2, &*itmon3 )
1072 || !_find( &*itmon3, &*itmon3 ) ) continue;
1073
1074 // add positive-definite cuts
1075 _MatPSD.push_back( _MatPSD[pos] );
1076 auto& mat3 = _MatPSD.back();
1077 assert( _insert( mat3, &*itmon3, &*itmon3, 1. )
1078 && _insert( mat3, &*itmon1, &*itmon3, -2. )
1079 && _insert( mat3, &*itmon2, &*itmon3, 2. ) );
1080 _MatPSD.push_back( _MatPSD[pos] );
1081 auto& mat4 = _MatPSD.back();
1082 assert( _insert( mat4, &*itmon3, &*itmon3, 1. )
1083 && _insert( mat4, &*itmon1, &*itmon3, 2. )
1084 && _insert( mat4, &*itmon2, &*itmon3, -2. ) );
1085 _MatPSD.push_back( _MatPSD[pos+1] );
1086 auto& mat5 = _MatPSD.back();
1087 assert( _insert( mat5, &*itmon3, &*itmon3, 1. )
1088 && _insert( mat5, &*itmon1, &*itmon3, -2. )
1089 && _insert( mat5, &*itmon2, &*itmon3, -2. ) );
1090 if( options.BASIS == Options::MONOM && !itmon1->gcexp()%2
1091 && !itmon2->gcexp()%2 && !itmon3->gcexp()%2 )
1092 continue;
1093 _MatPSD.push_back( _MatPSD[pos+1] );
1094 auto& mat6 = _MatPSD.back();
1095 assert( _insert( mat6, &*itmon3, &*itmon3, 1. )
1096 && _insert( mat6, &*itmon1, &*itmon3, 2. )
1097 && _insert( mat6, &*itmon2, &*itmon3, 2. ) );
1098 }
1099 }

1177template <typename KEY, typename COMP>
1178inline std::multimap<double,typename SQuad<KEY,COMP>::map_SPoly>
1179SQuad<KEY,COMP>::factorize
1180( map_SQuad const& mat )
1181const
1182{
1183 // Populate sparse symmetric coefficient matrix
1184 CPPL::dssmatrix coefmat;
1185 std::map< t_SMon, unsigned, lt_SMon<COMP> > indexmap;
1186 unsigned index = 0;
1187 for( auto const& [ijmon,coef] : mat ){
1188 auto itimon = indexmap.find( *ijmon.first );
1189 if( itimon == indexmap.end() ){
1190 itimon = indexmap.insert( std::make_pair( *ijmon.first, index++ ) ).first;
1191 coefmat.stretch( 1 );
1192 }
1193 // Diagonal entry
1194 if( ijmon.first == ijmon.second ){
1195 coefmat.put( itimon->second, itimon->second, coef );
1196 }
1197 // Off-diagonal entry
1198 else{
1199 auto itjmon = indexmap.find( *ijmon.second );
1200 if( itjmon == indexmap.end() ){
1201 itjmon = indexmap.insert( std::make_pair( *ijmon.second, index++ ) ).first;
1202 coefmat.stretch( 1 );
1203 }
1204 coefmat.put( itimon->second, itjmon->second, coef/2e0 );
1205 }
1206 }
1207
1208 // Perform eigenvalue decomposition
1209 std::multimap<double,map_SPoly> eigdec;
1210 std::vector<double> eigval;
1211 std::vector<CPPL::dcovector> eigvec;
1212 CPPL::dsymatrix dcoefmat = coefmat.to_dsymatrix();
1213 if( dcoefmat.dsyev( eigval, eigvec ) ) return eigdec;
1214
1215 // Populate decomposition map
1216 auto itval = eigval.begin();
1217 auto itvec = eigvec.begin();
1218 for( ; itval != eigval.end(); ++itval, ++itvec ){
1219 map_SPoly eigterm;
1220 for( auto const& [mon,index] : indexmap ){
1221 //if( isequal( itvec->array[index], 0. ) ) continue;
1222 eigterm[mon] = itvec->array[index];
1223 }
1224 eigdec.insert( std::make_pair( *itval, eigterm ) );
1225 }
1226
1227#ifdef MC__SQUAD_DEBUG_SEPARATE
1228 unsigned i = 0;
1229 for( auto const& [eigval,eigterm] : eigdec ){
1230 std::cout << std::endl << " Eigen-direction #" << ++i << ": " << std::scientific
1231 << std::right << std::setw(options.DISPLEN+7) << eigval << std::endl;
1232 for( auto const& [mon,coord] : eigterm )
1233 std::cout << " " << mon.display(options.BASIS) << ": "
1234 << std::right << std::setw(options.DISPLEN+7) << coord << std::endl;
1235 }
1236#endif
1237 return eigdec;
1238}

1549 for( auto const& mat : _MatFct ){

1552 auto const& [itmon,ins] = _MIP_SetMon.insert( *ijmon.first + *ijmon.second );

1555 }
1556 unsigned const maxDeg = _MIP_SetMon.rbegin()->tord;

1558 unsigned const nVar = _MIP_VarDeg.expr.size(); // Number of participating variables

1583 else{
1584 _MIP_auxexp[k].push_back( _GRBmodel->addVar( 0., maxExp, 0., GRB_SEMICONT ) );

1586 }
