334 switch( _ierr ){
335 case DIVZERO:
336 return "mc::SPoly\t Scalar division by zero";
337 case DIVPOLY:
338 return "mc::SPoly\t Division between two polynomials";
339 //case WRONGBASIS:

341 case INTERNAL:

343 return "mc::SPoly\t Internal error";
344 }
345 }

347 TYPE _ierr;

499 ++itmon;

503template <typename KEY, typename COMP>
504inline void
505SPoly<KEY,COMP>::_pow2cheb
506( std::map<unsigned,t_poly>& spmap )
507{
508 // Converts COEFF in monomial basis to the Chebyshev basis.
509 // Based on SCONMC function (http://www.netlib.org/math/MATH77/dconmc.f, http://www.netlib.org/math/docpdf/ch11-03.pdf)
510 int const N = spmap.crbegin()->first;
511 if( N <= 1 ) return;
512 // TP = .5D0**(N-1)
513 // COEFF(N) = TP * COEFF(N)
514 // COEFF(N-1) = TP * COEFF(N-1)
515 double TP = std::pow(0.5,N-1);
516 for( auto& [mon,coef] : spmap[N] ) coef *= TP;
517 for( auto& [mon,coef] : spmap[N-1] ) coef *= TP;
518 // do 20 J = N-2, 0, -1
519 // TP = 2.D0 * TP
520 // COEFF(J) = TP * COEFF(J)
521 // COEFF(J+1) = 2.D0 * COEFF(J+1)
522 for( int J=N-2; J>=0; --J ){
523 TP *= 2e0;
524 for( auto& [mon,coef] : spmap[J] ) coef *= TP;
525 for( auto& [mon,coef] : spmap[J+1] ) coef *= 2e0;
526 // do 10 I = J, N-2
527 // COEFF(I) = COEFF(I) + COEFF(I+2)
528 // 10 continue
529 for( int I=J; I<=N-2; ++I ){
530 for( auto& [mon,coef] : spmap[I+2] ){
531 auto [itmon,ins] = spmap[I].insert( std::make_pair( mon, coef ) );
532 if( !ins ) itmon->second += coef;
533 if( isequal( itmon->second, 0. ) ) spmap[I].erase( itmon );
534 }
535 }
536 // 20 continue
537 }
538 // return
539 // end
540}

542template <typename KEY, typename COMP>
543inline void
544SPoly<KEY,COMP>::_cheb2pow
545( std::map<unsigned,t_poly>& spmap )
546{
547 // Converts COEFF in Chebyshev basis to the monomial basis.
548 // Based on SCONCM function (http://www.netlib.org/math/MATH77/dconcm.f, http://www.netlib.org/math/docpdf/ch11-03.pdf)
549 int const N = spmap.crbegin()->first;
550 if( N <= 1 ) return;
551 // TP = 1.D0
552 double TP = 1e0;
553 // do 20 J = 0, N-2
554 for( int J=0; J<=N-2; ++J ){
555 // do 10 I = N-2, J, -1
556 // COEFF(I) = COEFF(I) - COEFF(I+2)
557 // 10 continue
558 for( int I=N-2; I>=J; --I ){
559 for( auto& [mon,coef] : spmap[I+2] ){
560 auto [itmon,ins] = spmap[I].insert( std::make_pair( mon, -coef ) );
561 if( !ins ) itmon->second -= coef;
562 if( isequal( itmon->second, 0. ) ) spmap[I].erase( itmon );
563 }
564 }
565 // COEFF(J+1) = .5D0 * COEFF(J+1)
566 // COEFF(J) = TP * COEFF(J)
567 // TP = 2.D0 * TP
568 // 20 continue
569 for( auto& [mon,coef] : spmap[J+1] ) coef /= 2e0;
570 for( auto& [mon,coef] : spmap[J] ) coef *= TP;
571 TP *= 2e0;
572 }
573 // COEFF(N) = TP * COEFF(N)
574 // COEFF(N-1) = TP * COEFF(N-1)
575 for( auto& [mon,coef] : spmap[N] ) coef *= TP;
576 for( auto& [mon,coef] : spmap[N-1] ) coef *= TP;
577 // return
578 // end
579}

581template <typename KEY, typename COMP>
582inline void
583SPoly<KEY,COMP>::_svec1Dfull
584( typename t_var::const_iterator itvar,
585 std::pair<t_mon,double> const& coefmon,
586 std::map<unsigned,t_poly>& mapspoly )
587{
588 auto& [mon,coef] = coefmon;
589 auto ie = mon.expr.find( *itvar );
590 if( ie == mon.expr.end() ) // no dependence on variable *itvar
591 mapspoly[ 0 ].insert( coefmon );
592 else{ // dependence on variable *itvar of order iord
593 auto& [ivar,iord] = *ie;
594 t_mon monmod( mon.tord - iord, mon.expr );
595 monmod.expr.erase( ivar ); // remove *itvar entry
596 mapspoly[ iord ].insert( std::make_pair( monmod, coef ) );
597 }
598}

609template <typename KEY, typename COMP>
610inline void
611SPoly<KEY,COMP>::_convert
612( t_poly& mapmon, t_var const& setvar, int const BASIS )
613const
614{
615 // Constant or linear polynomial - nothing to do
616 if( mapmon.empty() || setvar.empty() || mapmon.crbegin()->first.tord <= 1 ) return;
617
618 // Apply conversion variable by variable
619 for( auto itvar=setvar.cbegin(); itvar!=setvar.cend(); ++itvar ){
620 // Isolate factors of variable *itvar and convert to desired BASIS
621 std::map<unsigned,t_poly> spmap;
622 for( auto const& mon : mapmon ) _svec1Dfull( itvar, mon, spmap );
623#ifdef MC__SPOLY_DEBUG_CONVERT
624 _sdisp1D( spmap, itvar, "before:" );
625#endif
626 switch( BASIS ){
627 case Options::CHEB: _pow2cheb( spmap ); break;
628 case Options::MONOM: _cheb2pow( spmap ); break;
629 }
630#ifdef MC__SPOLY_DEBUG_CONVERT
631 _sdisp1D( spmap, itvar, "after:" );
632#endif
633 // Merge back into multivariate polynomial
634 mapmon.clear();
635 for( auto&& [iord,spoly] : spmap ){
636 if( !iord ){
637 mapmon = spmap[0];
638 continue;
639 }
640 for( auto&& [mon,coef] : spoly ){
641 t_mon monmod( mon.tord + iord, mon.expr );
642 monmod.expr[ *itvar ] = iord;
643 mapmon[ monmod ] = coef;
644 }
645 }
646#ifdef MC__SPOLY_DEBUG_CONVERT
647 std::cout << display( mapmon, options.BASIS, options.DISPLEN );
648#endif
649 //_clean( mapmon );
650 }
651}

719template <typename KEY, typename COMP>
720inline
721std::string
722SPoly<KEY,COMP>::display
723( t_poly mapmon, int const& BASIS, int const& DISPLEN, bool const ONELINE )
724const
725{
726 std::ostringstream out;
727 if( !ONELINE ) out << std::endl;
728 out << std::scientific << std::setprecision(DISPLEN);
729 bool first = true;
730 for( auto const& [mon,coef] : mapmon ){
731 if( ONELINE ){
732 if( first ){
733 first = false;
734 out << coef;
735 }
736 else if( coef < 0. )
737 out << " - " << -coef;
738 else if( coef > 0. )
739 out << " + " << coef;
740 if( mon.tord )
741 out << " " << mon.display( BASIS );
742 }
743 else
744 out << std::right << std::setw(DISPLEN+7) << coef << " " << std::setw(2) << mon.tord << " "
745 << mon.display( BASIS ) << std::endl;
746 }
747 return out.str();
748}

1041template <typename KEY, typename COMP>
1042inline void
1043SPoly<KEY,COMP>::_sprod1D
1044( std::map<unsigned,t_poly> const& sp1map,
1045 std::map<unsigned,t_poly> const& sp2map,
1046 t_poly& mapmon, t_var const& setvar,
1047 typename t_var::const_iterator itvar )
1048const
1049{
1050 // construct product matrix of polynomial coefficients
1051 auto itvarnext = itvar;
1052 if( !setvar.empty() ) std::advance( itvarnext, 1 );
1053 std::map<std::pair<unsigned,unsigned>,t_poly> sp12map;
1054 for( auto& [ndx1,poly1] : sp1map ){
1055 // empty monomial in sp1
1056 if( poly1.empty() ) continue;
1057 // constant monomial in sp1
1058 if( poly1.size() == 1 && !poly1.begin()->first.tord ){
1059 for( auto& [ndx2,poly2] : sp2map ){
1060 auto [it12,ins] = sp12map.insert( std::make_pair( std::make_pair(ndx1,ndx2), t_poly() ) );
1061 assert( ins ); // map is initially empty
1062 _sscal1D( poly2, poly1.begin()->second, it12->second );
1063 }
1064 continue;
1065 }
1066 // general monomial in sp1
1067 for( auto& [ndx2,poly2] : sp2map ){
1068 // empty monomial in sp2
1069 if( poly2.empty() ) continue; // no term
1070 // constant monomial in sp2
1071 if( poly2.size() == 1 && !poly2.begin()->first.tord ){
1072 auto [it12,ins] = sp12map.insert( std::make_pair( std::make_pair(ndx1,ndx2), t_poly() ) );
1073 assert( ins ); // map is initially empty
1074 _sscal1D( poly1, poly2.begin()->second, it12->second );
1075 continue;
1076 }
1077#ifdef MC__SPOLY_DEBUG_SPROD
1078 std::cout << "Term (" << ndx1 << "," << ndx2 << "):\n";
1079#endif
1080 std::map<unsigned,t_poly> sp11map, sp22map;
1081 for( auto& mon1 : poly1 )
1082 _svec1D( itvarnext, mon1, sp11map );
1083#ifdef MC__SPOLYEXPR_DEBUG_SPROD
1084 _sdisp1D( sp11map, itvarnext, "Poly #1: " );
1085#endif
1086 for( auto& mon2 : poly2 )
1087 _svec1D( itvarnext, mon2, sp22map );
1088#ifdef MC__SPOLYEXPR_DEBUG_SPROD
1089 _sdisp1D( sp22map, itvarnext, "Poly #2: " );
1090#endif
1091 auto [it12,ins] = sp12map.insert( std::make_pair( std::make_pair(ndx1,ndx2), t_poly() ) );
1092 assert( ins ); // map is initially empty
1093 _sprod1D( sp11map, sp22map, it12->second, setvar, itvarnext );
1094 }
1095 }
1096
1097 // construct 1D product result and augment remainder as appropriate
1098 mapmon.clear();
1099 for( auto const& [ndx12,poly12] : sp12map ){
1100 auto const& [ndx1,ndx2] = ndx12;
1101 // Product involving two constant terms
1102 if( !ndx1 && !ndx2 )
1103 _slift1D( poly12, 1., mapmon );
1104 // Product involving exactly one constant term
1105 else if( !ndx1 || !ndx2 )
1106 _slift1D( poly12, 1., mapmon, itvar, ndx1+ndx2 );
1107 // Product between non-constant terms
1108 else{
1109 switch( options.BASIS ){
1110 // Chebyshev basis functions
1111 case Options::CHEB:
1112 _slift1D( poly12, .5, mapmon, itvar, ndx1+ndx2 );
1113 if( ndx1 == ndx2 )
1114 _slift1D( poly12, .5, mapmon );
1115 else if( ndx1 > ndx2 )
1116 _slift1D( poly12, .5, mapmon, itvar, ndx1-ndx2 );
1117 else
1118 _slift1D( poly12, .5, mapmon, itvar, ndx2-ndx1 );
1119 break;
1120 // Power basis functions
1121 case Options::MONOM:
1122 _slift1D( poly12, 1., mapmon, itvar, ndx1+ndx2 );
1123 break;
1124 }
1125 }
1126 }
1127
1128 // clean sparse polynomial
1129 _clean( mapmon );
1130#ifdef MC__SPOLY_DEBUG_SPROD
1131 _sdisp1D( mapmon, "Prod: " );
1132#endif
1133}
