371void
372SElimBase::_MIP_optimize
373( Options const& opt )
374{
375 _MIPexcpt = false;
376 options = opt;
377
378 try{
379 // Run MIP optimization for a minimal representation
380 _MIP_reset();
381 _MIP_encode();

407inline
408void
409SElimBase::_MIP_encode
410()
411{
412 // Define Gurobi continuous and binary variables
413 for( auto const& [v,vset] : _mapVarElVar ){
414 // This also defines the objective function
415 _MIP_var[v] = _GRBmodel->addVar( 0., 1., _VarWeight[v], GRB_BINARY );
416 _MIP_varMTZ[v] = _GRBmodel->addVar( 0., _ndxVar.size()-1., 0., GRB_CONTINUOUS );
417 for( auto const& vv : vset )
418 _MIP_varvar[v][vv] = _GRBmodel->addVar( 0., 1., 0., GRB_BINARY );
419 }
420
421 for( auto const& [e,vset] : _mapCtrVarCand ){
422 _MIP_ctr[e] = _GRBmodel->addVar( 0., 1., 0., GRB_BINARY );
423#ifdef MC__SELIM_CHECK
424 assert( !_mapCtrVarCand[e].empty() );
425#endif
426 for( auto const& v : vset ){
427#ifdef MC__SELIM_DEBUG_MIP
428 std::cout << "Define: _MIP_ctrvar[" << e << "][" << v << "]\n";
429#endif
430 _MIP_ctrvar[e][v] = _GRBmodel->addVar( 0., 1., 0., GRB_BINARY );
431 }
432 }
433
434 for( auto const& [v,eset] : _mapVarCtrCand ){
435 GRBLinExpr sumctrvar;
436 for( auto const& e : eset ){
437 sumctrvar += _MIP_ctrvar[e][v];
438#ifdef MC__SELIM_DEBUG_MIP
439 std::cout << "Use: _MIP_ctrvar[" << e << "][" << v << "]\n";
440#endif
441 }
442 _GRBmodel->addConstr( sumctrvar, GRB_EQUAL, _MIP_var[v] );
443 }
444
445 for( auto const& [v,vset] : _mapVarElVar ){
446 for( auto const& vv : vset )
447 _GRBmodel->addConstr( _MIP_varvar[v][vv], GRB_LESS_EQUAL, _MIP_var[v] );
448 }
449
450 for( auto const& [e,vset] : _mapCtrVarCand ){
451 GRBLinExpr sumctrvar;
452 for( auto const& v : vset ){
453#ifdef MC__SELIM_DEBUG_MIP
454 std::cout << "Use: _MIP_ctrvar[" << e << "][" << v << "]\n";
455#endif
456 sumctrvar += _MIP_ctrvar[e][v];
457 }
458 _GRBmodel->addConstr( sumctrvar, GRB_EQUAL, _MIP_ctr[e] );
459 }
460
461 for( auto const& [e,vset] : _mapCtrVarCand ){
462 for( auto const& v : vset ){
463#ifdef MC__SELIM_DEBUG_MIP
464 std::cout << "Use: _MIP_ctrvar[" << e << "][" << v << "]\n";
465#endif
466 for( auto const& vv : _mapVarElVar[v] ){
467 if( _mapCtrVar[e].find( vv ) == _mapCtrVar[e].end() )
468 _GRBmodel->addConstr( 1. - _MIP_varvar[v][vv], GRB_GREATER_EQUAL, _MIP_ctrvar[e][v] );
469 else
470 _GRBmodel->addConstr( _MIP_varvar[v][vv], GRB_GREATER_EQUAL, _MIP_ctrvar[e][v] );
471 }
472 }
473 }
474
475 // MTZ subtour elimination
476 for( auto const& [v,vset] : _mapVarElVar ){
477 for( auto const& vv : vset )
478 _GRBmodel->addConstr( _MIP_varMTZ[v] - _MIP_varMTZ[vv], GRB_LESS_EQUAL,
479 _ndxVar.size() * (1.-_MIP_varvar[v][vv]) - 1. );
480 }
481}

710template <typename... ExtOps>
711inline
712std::ostream&
713operator<<
714( std::ostream& out, SElimEnv<ExtOps...> const& env)
715{
716 auto const& [vvar,vctr,vaux] = env._VarElim;
717 std::cout << std::endl
718 << vvar.size() << " VARIABLES MAY BE ELIMINATED" << std::endl;
719
720 if( vaux.empty() ){
721 for( auto itvar = vvar.begin(), itctr = vctr.begin();
722 itvar != vvar.end(); ++itvar, ++itctr )
723 out << "VARIABLE " << *itvar << " USING CONSTRAINT" << *itctr << "=0";
724 return out;
725 }
726
727 for( auto itvar = vvar.begin(), itctr = vctr.begin(), itaux = vaux.begin();
728 itvar != vvar.end(); ++itvar, ++itctr, ++itaux ){
729 std::ostringstream ext;
730 ext << " OF " << *itvar << " USING " << *itctr << "=0";
731 env._dag->output( env._dag->subgraph( 1, &*itaux ), ext.str(), out );
732 }
733 return out;
734}

762template <typename... ExtOps>
763inline
764void
765SElimEnv<ExtOps...>::process
766( std::set<unsigned> const& ndxCtr, FFVar const* pCtr,
767 std::map<FFVar const*,double,lt_FFVar> const& wVar,
768 bool const add2dag )
769{
770 if( ndxCtr.empty() ) return; // Nothing to do!
771 std::vector<FFVar> vCtr;
772 vCtr.reserve( ndxCtr.size() );
773 for( unsigned const& i : ndxCtr ) vCtr.push_back( pCtr[i] );
774 process( ndxCtr.size(), vCtr.data(), wVar, add2dag );
775}
776
777template <typename... ExtOps>
778inline
779void
780SElimEnv<ExtOps...>::process
781( unsigned const nCtr, FFVar const* pCtr,
782 std::map<FFVar const*,double,lt_FFVar> const& wVar,
783 bool const add2dag )
784{
785 // Reset internal variables
786 _reset();
787
788 // Pass options to base class
789 SElimBase::options = options;
790
791 // Update participating variables in _Var
792 auto sgCtr = _dag->subgraph( nCtr, pCtr );
793 unsigned v=0;
794 for( auto const& Op : sgCtr.l_op ){
795 if( Op->type != FFOp::VAR ) continue;
796 _Var.push_back( *Op->varout[0] );
797 _DVar.push_back( FFDep().indep( Op->varout[0]->id().second ) );
798 _IVar.push_back( FFInv().indep( v ) );
799 auto itv = wVar.find( Op->varout[0] );
800 _VarWeight[v] = ( itv != wVar.end() ? itv->second : 1. );
801 ++v;
802 }
803#ifdef MC__SELIM_DEBUG_PROCESS
804 std::cout << std::endl << _Var.size() << " Original Variables: ";
805 for( auto const& var : _Var ) std::cout << var << " ";
806 std::cout << std::endl;
807#endif
808
809 // Find out variable invertibility in equality constraints
810 _Ctr.assign( pCtr, pCtr+nCtr );
811 _ICtr.resize( nCtr );
812 FFInv::options.INVOP = options.ELIMNLIN;
813 //std::cout << "FFInv::options.INVOP: " << FFInv::options.INVOP.size() << std::endl;
814 _dag->eval( sgCtr, nCtr, pCtr, _ICtr.data(), _Var.size(), _Var.data(), _IVar.data() );
815
816 // Create variable multiplicity map !!!THIS IS NOT TRULY MULTIPLICITY - MULTIPLE OCCURENCES COULD BE WITHIN SINGLE FUNCTION
817 for( unsigned j=0; j<_ICtr.size(); ++j )
818 for( auto const& [i,type] : _ICtr[j].inv() )
819 if( _mapVarMult.count(i) ) _mapVarMult[i]++;
820 else _mapVarMult[i] = 1;
821#ifdef MC__SELIM_DEBUG_PROCESS
822 std::cout << std::endl << "Participating variable multiplicity:" << std::endl;
823 for( auto const& [i,m] : _mapVarMult ){
824 std::cout << i << " (" << _Var[i] << "):" << m;
825 std::cout << std::endl;
826 }
827#endif
828
829 // Create candidate variable and constraint sets and maps
830 for( unsigned j=0; j<_ICtr.size(); ++j ){
831#ifdef MC__SELIM_DEBUG_PROCESS
832 std::cout << j << ": " << _ICtr[j] << std::endl;
833#endif
834 bool isinvert = false;
835 for( auto const& [i,type] : _ICtr[j].inv() ){
836 // Exclude variables based on weight and/or multiplicity
837 if( _VarWeight[i] <= 0.
838 || (options.MULTMAX > 0 && _mapVarMult[i] > options.MULTMAX) ) continue;
839 switch( type ){
840 case FFInv::L:
841 if( options.ELIMLIN ){
842 _ndxVar.insert( i );
843 isinvert = true;
844 }
845 break;
846 case FFInv::S:
847 if( options.ELIMMLIN ){
848 _ndxVar.insert( i );
849 isinvert = true;
850 }
851 break;
852 case FFInv::N:
853 _ndxVar.insert( i );
854 isinvert = true;
855 break;
856 case FFInv::U:
857 default:
858 break;
859 }
860 }
861 if( isinvert ) _ndxCtr.insert( j );
862 }
863#ifdef MC__SELIM_DEBUG_PROCESS
864 std::cout << std::endl << "Candidate variables for elimination:";
865 for( unsigned const& v : _ndxVar ) std::cout << " " << v;
866 std::cout << std::endl;
867 std::cout << std::endl << "Candidate constraints for elimination:";
868 for( unsigned const& e : _ndxCtr ) std::cout << " " << e;
869 std::cout << std::endl;
870#endif
871
872 for( unsigned const& i : _ndxVar )
873 _mapVarElVar[i] = std::set<unsigned>();
874 for( unsigned const& j : _ndxCtr ){
875 for( auto const& [i,type] : _ICtr[j].inv() ){
876 if( _ndxVar.find( i ) == _ndxVar.end() ) continue;
877 _mapCtrVar[j].insert( i );
878 bool insert = false;
879 switch( type ){
880 case FFInv::L: if( options.ELIMLIN ) insert = true; break;
881 case FFInv::S: if( options.ELIMMLIN ) insert = true; break;
882 case FFInv::N: insert = true; break;
883 case FFInv::U: default: break;
884 }
885 if( insert ){
886 _mapCtrVarCand[j].insert( i );
887 _mapVarCtrCand[i].insert( j );
888 for( auto const& [k,type] : _ICtr[j].inv() )
889 if( k != i && _ndxVar.find( k ) != _ndxVar.end() )
890 _mapVarElVar[i].insert( k );
891 }
892 }
893 }
894#ifdef MC__SELIM_DEBUG_PROCESS
895 std::cout << std::endl << "Participating variables in each constraint:" << std::endl;
896 for( auto const& [e,ndx] : _mapCtrVar ){
897 std::cout << e << " (" << _Ctr[e] << "):";
898 for( auto const& v : ndx ) std::cout << " " << v;
899 std::cout << std::endl;
900 }
901 std::cout << std::endl << "Candidate variables in each constraint:" << std::endl;
902 for( auto const& [j,ndx] : _mapCtrVarCand ){
903 std::cout << j << " (" << _Ctr[j] << "):";
904 for( auto const& i : ndx ) std::cout << " " << i;
905 std::cout << std::endl;
906 }
907 std::cout << std::endl << "Candidate variables to substitute each variable:" << std::endl;
908 for( auto const& [i,ndx] : _mapVarElVar ){
909 std::cout << i << " (" << _Var[i] << "):";
910 for( auto const& k : ndx ) std::cout << " " << k << " (" << _Var[k] << ")";
911 std::cout << std::endl;
912 }
913 std::cout << std::endl << "Candidate constraints for variable elimination:" << std::endl;
914 for( auto const& [i,ndx] : _mapVarCtrCand ){
915 std::cout << i << " (" << _Var[i] << "):";
916 for( auto const& j : ndx ) std::cout << " " << j;
917 std::cout << std::endl;
918 }
919#endif
920
921 // Determine an optimal elimination set
922#if defined(MC__USE_GUROBI)
923 _MIP_optimize( options );
924 if( _MIPexcpt ) throw Exceptions( Exceptions::MIPERR );
925 _MIP_decode();
926#else
927 throw Exceptions( Exceptions::MIPERR );
928#endif
929
930 // Decode optimization results
931 auto& [vvar,vctr,vaux] = _VarElim;
932#ifdef MC__SELIM_DEBUG_PROCESS

976 // Case multiple dependents

1030 }

1032 }

1042 }
1043
1044 // Split numerator into dependent and independent parts
1045 FFVar const* pdep = *sdep.cbegin();
1046 t_mon mondep( pdep );
1047 unsigned orddep = 0;
1048 t_poly polydep, polyindep;
1049 for( auto const& [mon,coef] : numer.mapmon() ){

1051 t_mon mondepord;
1052 for( unsigned q=0; (mondepord+mondep).subseteq( mon ); ++q )
1053 mondepord += mondep;

1055 if( orddep && mondepord.tord && orddep != mondepord.tord ) throw Exceptions( Exceptions::INVERT );

1057 if( mondepord.tord )
1058 polydep += std::make_pair( mon - mondepord, coef );
1059 else
1060 polyindep -= std::make_pair( mon, coef );
1061 }

1063 std::cout << "Dependent variable: " << *pdep << "^" << orddep << std::endl;
1064 std::cout << "Dependent part:" << polydep;
1065 std::cout << "Independent part:" << polyindep;
1066#endif

1069 if( pdep->id().first == FFVar::VAR ){
1070 // Create new DAG expression and copy pointer to expression in _VarElim
1071 *itaux = *_insert_expr( VarEl, polyindep, polydep, false, orddep );
1072#ifdef MC__SELIM_DEBUG_PROCESS
1073 std::cout << "INSERTED DAG EXPRESSION:";
1074 std::ostringstream ext;
1075 ext << " OF " << *pdep;
1076 _dag->output( _dag->subgraph( 1, &*itaux ), ext.str() );
1077#endif
1078 break; // EXIT THE INNER FOR LOOP ON INTERMEDIATE VARIABLES
1079 }
1080
1081 // Case dependent is not a DAG variable, identify intermediate operation
1082 bool found_Op = false;
1083 for( auto const& [pOp,SPVar] : _Interm ){
1084 if( pOp->varout[0]->id().first != FFVar::AUX ) throw Exceptions( Exceptions::INVERT );
1085 if( pOp->varout[0]->id().second != pdep->id().second ) continue;
1086 found_Op = true;
1087 // Create new DAG expression and update inverted expression
1088 auto pexpr = _insert_expr( ndxVarEl, VarEl, SPVar, pOp, polyindep, polydep, false, orddep );
1089 *itaux = *pexpr.first;
1090 spvar = pexpr.second;
1091#ifdef MC__SELIM_DEBUG_PROCESS
1092 std::cout << "Next dependent variable: " << *spvar << std::endl;
1093#endif
1094 break;
1095 }
1096 if( !found_Op ) throw Exceptions( Exceptions::INVERT );
1097 }
1098 }

1190 ndxsp = 1;

1195 case FFOp::CHEB:

1222template <typename... ExtOps>
1223inline
1224FFVar const*
1225SElimEnv<ExtOps...>::_insert_expr
1226( FFVar& var, t_poly const& polyindep, t_poly const& polydep, bool const useprod,
1227 int const order )
1228{
1229 //std::cout << "ENTERING _insert_expr\n";
1230 if( !order ) throw Exceptions( Exceptions::INVERT );
1231 bool const dagaux = true;
1232
1233 // Case inverse operation is needed
1234 if( order < 0 ){
1235 if( !polyindep.mapmon().empty() ){
1236 var += insert_dag( polyindep, useprod, dagaux );
1237 }
1238 var = insert_dag( polydep, useprod, dagaux ) / var;
1239 // Variable has order greater than 1
1240 if( order < -1 ) var = pow( var, -1./(double)order );
1241 }
1242
1243 // Case dependent term is constant
1244 else if( !polydep.maxord() ){
1245 double const cst = polydep.coefmon( t_mon() );
1246 // Constant dependent term is one
1247 if( cst == 1e0 ){
1248 if( !polyindep.mapmon().empty() )
1249 var += insert_dag( polyindep, useprod, dagaux );
1250 }
1251 // Constant dependent term is negative one
1252 else if( cst == -1e0 ){
1253 if( !polyindep.mapmon().empty() )
1254 var = insert_dag( - polyindep, useprod, dagaux ) - var;
1255 else
1256 var = -var;
1257 }
1258 else{
1259 if( !polyindep.mapmon().empty() )
1260 var += insert_dag( polyindep, useprod, dagaux );
1261 var /= cst;
1262 }
1263 // Variable has order greater than 1
1264 if( order > 1 ) var = pow( var, 1./(double)order );
1265 }
1266
1267 // Case dependent term is monomial
1268 else if( polydep.nmon() == 1 && order == 1 ){
1269
1270 // Split numerator between reduced and non-reduced parts first
1271 auto const& [mondep,coefdep] = *polydep.mapmon().cbegin();
1272 t_poly polyindepred, polyindepnred;
1273 for( auto const& [mon,coef] : polyindep.mapmon() ){
1274 if( mondep.subseteq( mon ) )
1275 polyindepred += std::make_pair( mon-mondep, coef/coefdep );
1276 else
1277 polyindepnred += std::make_pair( mon, coef );
1278 }
1279 if( !polyindepnred.mapmon().empty() )
1280 var += insert_dag( polyindepnred, useprod, dagaux );
1281 var /= insert_dag( polydep, useprod, dagaux );
1282 if( !polyindepred.mapmon().empty() )
1283 var += insert_dag( polyindepred, useprod, dagaux );
1284 }
1285
1286 // Case dependent term is linear
1287 // General case
1288 else{
1289 var += insert_dag( polyindep, useprod, dagaux );
1290 var /= insert_dag( polydep, useprod, dagaux );
1291 // Variable has order greater than 1
1292 if( order > 1 ) var = pow( var, 1./(double)order );
1293 }
1294
1295 // Return pointer to internal DAG variable
1296 return _ptr_expr( var );
1297}
