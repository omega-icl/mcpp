312 public:
314 enum TYPE{
315 DIV=1,
316 INV,
317 LOG,
318 SQRT,
319 TAN,
320 ACOS,
321 EIGEN,
322 INIT=-1,
323 INCON=-2,
324 CMODEL=-3,
325 INTERNAL = -4,
326 UNDEF=-33
327 };
328
329 Exceptions( TYPE ierr ) : _ierr( ierr ){}
331 int ierr(){ return _ierr; }
333 std::string what(){
334 switch( _ierr ){
335 case DIV:
336 return "mc::CModel\t Division by zero scalar";
337 case INV:
338 return "mc::CModel\t Inverse operation with zero in range";
339 case LOG:
340 return "mc::CModel\t Log operation with non-positive numbers in range";
341 case SQRT:
342 return "mc::CModel\t Square-root operation with negative numbers in range";
343 case TAN:
344 return "mc::SCModel\t Tangent operation with (k+1/2)·PI in range";
345 case ACOS:
346 return "mc::CModel\t Sine/Cosine inverse operation with range outside [-1,1]";
347 case EIGEN:
348 return "mc::CModel\t Range bounder with eigenvalue decomposition failed";
349 case INIT:
350 return "mc::CModel\t Chebyshev variable initialization failed";
351 case INCON:
352 return "mc::CModel\t Inconsistent bounds with template parameter arithmetic";
353 case CMODEL:
354 return "mc::CModel\t Operation between Chebyshev variables in different Chebyshev model environment not allowed";
355 case UNDEF:
356 return "mc::CModel\t Feature not yet implemented in mc::CModel class";
357 case INTERNAL:
358 return "mc::CModel\t Internal error";
359 default:
360 return "mc::CModel\t Undocumented error";
361 }
362 }
363
364 private:
365 TYPE _ierr;
366 };

1003template <typename T> inline void
1004CModel<T>::_size
1005()
1006{
1007#ifndef MC__CVAR_SPARSE_PRODUCT_NAIVE
1008 if( !_sparse ) _set_prodmon();
1009 else _prodmon = 0;
1010#else
1011 _set_prodmon();
1012#endif
1013 _bndpow = new T*[_nvar];
1014 for( unsigned i=0; i<_nvar; i++ ) _bndpow[i] = 0;
1015 _bndmon = new T[_nmon];
1016 _bndvar = new T[_nvar];
1017 for( unsigned i=0; i<_nvar; i++ ) _bndvar[i] = 0.;
1018 _coefinterp.resize( _nord+options.INTERP_EXTRA+1 );
1019
1020 _CV = new CVar<T>( this );
1021}

1023template <typename T> inline CModel<T>*
1024CModel<T>::lift
1025( const unsigned nx, const T*X ) const
1026{
1027 CModel<T>* CM = new CModel<T>( _nvar+nx, _nord );
1028 unsigned ix = 0;
1029 for( ; ix<_nvar; ix++ )
1030 CVar<T> var( CM, ix, _bndvar[ix] );
1031 for( ; ix<nx; ix++ )
1032 CVar<T> var( CM, ix, X[ix-_nvar] );
1033 return CM;
1034}

1051 delete[] _prodmon[i];
1052 }

1071{

1081

1085

1087 _prodmon[imon] = new unsigned int*[imon+1];

1089 for( ; jmon<_posord[jord+1] && jmon<=imon; jmon++ ){

1092 iexp[0] = 0; // used to store partial order

1100 unsigned nprodmax = 1, nprodmon = 1, kvar = 0;
1101 for( unsigned ivar=0; ivar<_nvar; ivar++ )
1102 if( iexp[ivar+1] && _expmon[jmon*_nvar+ivar] ) nprodmax*=2;
1103 _set_prodmon_exp( kvar, _expmon+jmon*_nvar, jord, iexp, nprodmon );
1104 _prodmon[imon][jmon] = new unsigned int[nprodmon+2];
1105 _prodmon[imon][jmon][0] = nprodmax;
1106 _prodmon[imon][jmon][1] = 0;
1107 for( unsigned kmon=0; kmon<nprodmon; kmon++ ){

1110#endif

1114 }

1134template <typename T> inline void
1135CModel<T>::_set_prodmon_exp
1136( unsigned int&kvar, const unsigned int*iexp, const unsigned iord,
1137 unsigned int*iprodexp, unsigned int&nprodmon )
1138{
1139 if( kvar == _nvar || !nprodmon ) return;
1140
1141 int nprodnew = 0;
1142 for( unsigned imon=0; imon<nprodmon; imon++ ){
1143 // Product with Chebyshev basis function T0
1144 if( !iexp[kvar] || !iprodexp[imon*(_nvar+1)+kvar+1] ){
1145 iprodexp[imon*(_nvar+1)] += iprodexp[imon*(_nvar+1)+kvar+1] + iexp[kvar];
1146 if( iprodexp[imon*(_nvar+1)] <= _nord )
1147 iprodexp[imon*(_nvar+1)+kvar+1] += iexp[kvar];
1148 }
1149 // Product with Chebyshev basis function Ti, i>0
1150 else{
1151 // Only add term if resulting monic index remain less than or equal to _nord
1152 if( iprodexp[imon*(_nvar+1)] + iprodexp[imon*(_nvar+1)+kvar+1] + iexp[kvar] <= _nord ){
1153 for( unsigned ivar=0; ivar<=_nvar; ivar++ )
1154 iprodexp[(nprodmon+nprodnew)*(_nvar+1)+ivar] = iprodexp[imon*(_nvar+1)+ivar];
1155 iprodexp[(nprodmon+nprodnew)*(_nvar+1)] += iprodexp[imon*(_nvar+1)+kvar+1] + iexp[kvar];
1156 iprodexp[(nprodmon+nprodnew)*(_nvar+1)+kvar+1] += iexp[kvar];
1157 nprodnew++;
1158 }
1159 if( iexp[kvar] <= iprodexp[imon*(_nvar+1)+kvar+1] ){
1160 iprodexp[imon*(_nvar+1)] += iprodexp[imon*(_nvar+1)+kvar+1] - iexp[kvar];
1161 if( iprodexp[imon*(_nvar+1)] <= _nord )
1162 iprodexp[imon*(_nvar+1)+kvar+1] -= iexp[kvar];
1163 }
1164 else{
1165 iprodexp[imon*(_nvar+1)] += iexp[kvar] - iprodexp[imon*(_nvar+1)+kvar+1];
1166 if( iprodexp[imon*(_nvar+1)] <= _nord )
1167 iprodexp[imon*(_nvar+1)+kvar+1] = iexp[kvar] - iprodexp[imon*(_nvar+1)+kvar+1];
1168 }
1169 }
1170 }
1171 nprodmon += nprodnew;
1172 return _set_prodmon_exp( ++kvar, iexp, iord, iprodexp, nprodmon );
1173}

1347template <typename T> inline void
1348CModel<T>::_interpolation
1349( std::vector<double>&coefmon, const unsigned nord, const T&X,
1350 puniv f )
1351{
1352 double b( Op<T>::mid(X) ), a( Op<T>::u(X)-b ), x[nord+1], fx[nord+1];
1353 double mulconst( PI/(2.*double(nord+1)) );
1354 for( unsigned i(0); i<=nord; i++ ){
1355 x[i] = std::cos(mulconst*(2.*double(i)+1.));
1356 fx[i] = f( a*x[i]+b );
1357 }
1358
1359 switch( nord ){
1360 case 0:
1361 coefmon[0] = fx[0];
1362 return;
1363 case 1:
1364 coefmon[0] = 0.5 * ( fx[0] + fx[1] );
1365 coefmon[1] = ( fx[1] - fx[0] ) / ( x[1] - x[0] );
1366 return;
1367 default:
1368 for( unsigned i(0); i<=nord; i++ ){
1369 double mulconst2( std::cos(mulconst*double(i)) ),
1370 mulconst3( 4*std::pow(mulconst2,2)-2 ),
1371 b0( 0 ), b1( 0 );
1372 b0 = fx[nord];
1373 b1 = fx[nord-1] + mulconst3*b0;
1374 for( unsigned j=nord-2; j>1; j-=2 ){
1375 b0 = fx[j] + mulconst3*b1 - b0;
1376 b1 = fx[j-1] + mulconst3*b0 - b1;
1377 }
1378 if( !(nord%2) )
1379 b0 = fx[0] + mulconst3*b1 - b0 - b1;
1380 else{
1381 b0 = fx[1] + mulconst3*b1 - b0;
1382 b0 = fx[0] + mulconst3*b0 - b1 - b0;
1383 }
1384 coefmon[i] = 2./double(nord+1)*mulconst2*b0;
1385 }
1386 coefmon[0] *=0.5;
1387 return;
1388 }

1541template <typename T> inline void
1542CModel<T>::_sprod1D
1543( const std::vector<std::map<unsigned,double>>&CV1map,

1550
1551 // construct product matrix of polynomial coefficients
1552 std::vector<std::map<unsigned,double>> CV12map( (_nord+1)*(_nord+1) );
1553 for( unsigned iord1=0; iord1<=_nord; iord1++ ){
1554 if( CV1map[iord1].empty() ) continue; // no term
1555 if( CV1map[iord1].size() == 1 && !CV1map[iord1].begin()->first ){ // constant term only
1556 for( unsigned iord2=0; iord2<=_nord; iord2++ )
1557 _sscal1D( CV2map[iord2], CV1map[iord1].begin()->second, CV12map[iord1*(_nord+1)+iord2] );

1561 if( CV2map[iord2].empty() ) continue; // no term

1563 _sscal1D( CV1map[iord1], CV2map[iord2].begin()->second, CV12map[iord1*(_nord+1)+iord2] );
1564 continue;

1566#ifdef MC__POLYMODEL_DEBUG_SPROD

1568#endif
1569 std::vector<std::map<unsigned,double>> CV11map( _nord+1 ), CV22map( _nord+1 ); // component maps for both coefficients
1570 for( auto it=CV1map[iord1].begin(); it!=CV1map[iord1].end(); ++it ){
1571 for( unsigned ivar=ndxvar+2; ivar<_nvar; ivar++ ) iexp[ivar] = expmon(it->first)[ivar];
1572 CV11map[ expmon(it->first)[ndxvar+1] ].insert( std::make_pair( _loc_expmon(iexp), it->second ) );
1573 }
1574#ifdef MC__POLYMODEL_DEBUG_SPROD
1575 _sdisp1D( CV11map, ndxvar+1, "Var #1: " );
1576#endif
1577 for( auto it=CV2map[iord2].begin(); it!=CV2map[iord2].end(); ++it ){
1578 for( unsigned ivar=ndxvar+2; ivar<_nvar; ivar++ ) iexp[ivar] = expmon(it->first)[ivar];
1579 CV22map[ expmon(it->first)[ndxvar+1] ].insert( std::make_pair( _loc_expmon(iexp), it->second ) );
1580 }
1581#ifdef MC__POLYMODEL_DEBUG_SPROD
1582 _sdisp1D( CV22map, ndxvar+1, "Var #2: " );
1583#endif
1584 _sprod1D( CV11map, CV22map, CV12map[iord1*(_nord+1)+iord2], coefrem, ndxvar+1 );
1585 }
1586 }
1587
1588 // construct result map and augment remainder as appropriate
1589 CVRmap.clear();
1590 for( unsigned l=0; l<=_nord; l++ )
1591 _slift1D( CV12map[l*(_nord+1)+l], l?1.:2., CVRmap );
1592 for( unsigned k=1; k<=_nord; k++ ){

1597 _slift1D( CV12map[(l+k)*(_nord+1)+l], 1., CVRmap, coefrem, ndxvar, k, iexp );
1598 }
1599 }
1600 for( unsigned k=_nord+1; k<=2*_nord; k++ )
1601 for( unsigned l=k-_nord; l<=_nord; l++ )
1602 _slift1D( CV12map[(k-l)*(_nord+1)+l], coefrem );
1603 // rescale the coefficients by 0.5
1604 for( auto it=CVRmap.begin(); it!=CVRmap.end(); ++it ) it->second *= 0.5;
1605#ifdef MC__POLYMODEL_DEBUG_SPROD
1606 _sdisp1D( CVRmap, ndxvar, "Prod: " );

1691 if( iexp[i1] == 2 ){
1692 Umat[_nvar*i1+i1] = 2.*Ci;
1693 bndpol -= Ci;
1694 continue;
1695 }
1696 for( i2=i1+1; i2<_nvar; i2++ ) if( iexp[i2] ) break;
1697 Umat[_nvar*i1+i2] = 0.;
1698 Umat[_nvar*i2+i1] = Ci/2.;

1743template <typename T> template <typename C, typename U> inline U
1744CModel<T>::_polybound_LSB
1745( const C*coefmon, const U*bndord, const U*const*bndbasis,
1746 const std::set<unsigned>&ndxmon ) const
1747{
1748 static const double TOL = 1e-8;
1749
1750 U bndpol = (ndxmon.empty() || ndxmon.find(0)!=ndxmon.end())? coefmon[0]: 0.;
1751 //U bndpol = coefmon[0];
1752 if( _nord == 1 ) bndpol += bndord[1];
1753
1754 else if( _nord > 1 ){
1755 for( unsigned i=_posord[2]; i<_posord[3]; i++ ){
1756 unsigned k1=_nvar, k2=_nvar;
1757 const unsigned*iexp = _expmon+i*_nvar;
1758 for( unsigned j=0; k2==_nvar; j++ ){
1759 if( iexp[j] == 2 ) k1 = k2 = j;
1760 else if( iexp[j] && k1 == k2 ) k1 = j;
1761 else if( iexp[j] ) k2 = j;
1762 }
1763 C Ci = (ndxmon.empty() || ndxmon.find(i)!=ndxmon.end())? coefmon[i]: 0.;
1764 C Ck = (ndxmon.empty() || ndxmon.find(_nvar-k1)!=ndxmon.end())? coefmon[_nvar-k1]: 0.;
1765 // off-diagonal quadratic terms
1766 if( k1 < k2 ){
1767 bndpol += Ci * bndbasis[k1][1] * bndbasis[k2][1];
1768 continue;
1769 }
1770 // linear and diagonal quadratic terms
1771 const double ai = Ck, aii = Ci;
1772 if( std::fabs(aii) > TOL )
1773 bndpol += (2.*aii)*Op<U>::sqr(bndbasis[k1][1]+ai/(aii*4.))-aii-ai*ai/8./aii;
1774 else
1775 bndpol += ai*bndbasis[k1][1] + aii*bndbasis[k1][2];
1776 }
1777 }
1778 // higher-order terms
1779 for( unsigned i=3; i<=_nord; i++ ) bndpol += bndord[i];
1780 return bndpol;
1781}
